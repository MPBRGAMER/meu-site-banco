    57→    57→          orderBy: { data: "desc" },

   152→   152→        const data = await db.priceReport.findMany({

   154→   154→          orderBy: { data: "desc" },
   155→   155→          take: 200,
   156→   156→        });
   157→   157→        return json(data);
   158→   158→      }

   160→   160→        const reports = await db.priceReport.findMany({
   161→   161→          orderBy: { data: "desc" },
   162→   162→        });
   163→   163→        const grouped: Record<string, { nickname: string; count: number; lastReport: string }> = {};
   164→   164→        for (const r of reports) {
   165→   165→          const key = r.nickname.toLowerCase();
   166→   166→          if (!grouped[key]) {
   167→   167→            grouped[key] = { nickname: r.nickname, count: 0, lastReport: r.data.toISOString() };
   168→   168→          }
   169→   169→          grouped[key].count++;
   170→   170→          if (r.data > new Date(grouped[key].lastReport)) {
   171→   171→            grouped[key].lastReport = r.data.toISOString();
   172→   172→          }
   173→   173→        }
   174→   174→        const ranking = Object.values(grouped)
   175→   175→          .sort((a, b) => b.count - a.count)
   176→   176→          .slice(0, 10);
   177→   177→        return json(ranking);

   183→   183→        return json({ slotId, codigo: ad?.codigo || "" });
   184→   184→      }
   185→   185→      case "getAllAds": {
   186→   186→        const ads = await db.propaganda.findMany();
   187→   187→        const map: Record<string, string> = {};
   188→   188→        for (const a of ads) {
   189→   189→          if (a.codigo) map[a.slotId] = a.codigo;
   190→   190→        }
   191→   191→        return json(map);
   192→   192→      }
   193→   193→      case "events": {
   194→   194→        // Server-Sent Events for real-time updates
   195→   195→        const stream = new ReadableStream({
   196→   196→          start(controller) {
   197→   197→            const interval = setInterval(() => {
   198→   198→              try {
   199→   199→                controller.enqueue(`data: ${JSON.stringify({ t: Date.now() })}\n\n`);
   200→   200→              } catch {
   201→   201→                clearInterval(interval);
   202→   202→                controller.close();
   203→   203→              }
   204→   204→            }, 5000);
   205→   205→            // Clean up on close
   206→   206→        req.signal.addEventListener("abort", () => {
   207→   207→              clearInterval(interval);
   208→   208→              controller.close();
   209→   209→            });
   210→   210→          },
   211→   211→        });
   212→   212→        return new Response(stream, {
   213→   213→          headers: {
   214→   214→            "Content-Type": "text/event-stream",
   215→   215→            "Cache-Control": "no-cache",
   216→   216→            "Connection": "keep-alive",
   217→   217→          },
   218→   218→        });
   219→   219→      }
   220→   220→      default:
   221→   221→        return err("Ação GET desconhecida: " + action);
   222→   222→    }
   223→   223→  } catch (e: unknown) {
   224→   224→    const msg = e instanceof Error ? e.message : "Erro interno";
   225→   225→    return err(msg, 500);
   226→   226→  }
   227→   227→}

   229→   229→export async function POST(req: NextRequest) {
   230→   230→  try {
   231→   231→    const body = await req.json();
   232→   232→    const { action, ...data } = body;
   233→   233→
   234→   234→    if (!action) return err("Campo 'action' é obrigatório");
   235→   235→
   236→   236→    // Ações públicas que não precisam de admin
   237→   237→    const PUBLIC_ACTIONS = new Set([
   238→   238→      "participarSorteio",
   239→   239→      "comprarNumero",
   240→   240→      "addLance",
   241→   241→      "reportPrice",
   242→   242→    ]);
   243→   243→
   244→   244→    // Todas as outras ações POST exigem autenticação de admin
   245→   245→    if (!PUBLIC_ACTIONS.has(action)) {
   246→   246→      const pwd = req.headers.get("x-admin-password");
   247→   247→      if (!pwd || pwd !== ADMIN_PASSWORD) return err("Não autorizado", 403);
   248→   248→    }
   249→   249→
   250→   250→    switch (action) {
   251→   251→      // === EMPRÉSTIMOS ===
   252→   252→      case "addEmprestimo": {
   253→   253→        const { player, item, quantidade, dataEmprestimo, tipoMembro, status } = data;
   254→   254→        if (!player || !item || !quantidade) return err("Campos obrigatórios: player, item, quantidade");
   255→   255→        const emp = await db.emprestimo.create({
   256→   256→          data: {
   257→   257→            player,
   258→   258→            item,
   259→   259→            quantidade: Number(quantidade),
   260→   260→            dataEmprestimo: dataEmprestimo ? new Date(dataEmprestimo) : new Date(),
   261→   261→            tipoMembro: tipoMembro || "comum",
   262→   262→            status: status || "pendente",
   263→   263→          },
   264→   264→        });
   265→   265→        return json(emp);
   266→   266→      }
   267→   267→      case "updateEmprestimo": {
   268→   268→        const { id, status, dataPagamento, itemPagamento, quantidadePaga } = data;
   269→   269→        if (!id) return err("id obrigatório");
   270→   270→        const updateData: Record<string, unknown> = {};
   271→   271→        if (status) updateData.status = status;
   272→   272→        if (dataPagamento) updateData.dataPagamento = new Date(dataPagamento);
   273→   273→        if (itemPagamento) updateData.itemPagamento = itemPagamento;
   274→   274→        if (quantidadePaga !== undefined) updateData.quantidadePaga = Number(quantidadePaga);
   275→   275→        const emp = await db.emprestimo.update({
   276→   276→          where: { id },
   277→   277→          data: updateData,
   278→   278→        });
   279→   279→        return json(emp);
   280→   280→      }
   281→   281→
   282→   282→      // === INVESTIDORES ===
   283→   283→      case "addInvestidor": {
   284→   284→        const { nome, observacao } = data;
   285→   285→        if (!nome) return err("nome obrigatório");
   286→   286→        const inv = await db.investidor.create({
   287→   287→          data: { nome, observacao: observacao || null, ordem: 0 },
   288→   288→        });
   289→   289→        return json(inv);
   290→   290→      }
   291→   291→      case "removeInvestidor": {

   293→   293→        if (!id) return err("id obrigatório");

   301→   301→          await db.investidor.update({
   302→   302→            where: { id: u.id },
   303→   303→            data: { ordem: u.ordem },

   312→   312→        if (!itemBase || !quantidadeBase || !itemResultado || !quantidadeResultado) {
   313→   313→          return err("Campos obrigatórios: itemBase, quantidadeBase, itemResultado, quantidadeResultado");
   314→   314→        }
