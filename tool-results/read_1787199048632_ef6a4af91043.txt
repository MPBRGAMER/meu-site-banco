  1349→  1398→function translatePage(langCode: string) {
  1350→  1399→  if (isTranslating) return;
  1351→  1400→  isTranslating = true;
  1352→  1401→  try {
  1353→  1402→    currentLang = langCode;
  1354→  1403→    const dictionary = dictionaries[langCode];
  1355→  1404→    document.documentElement.lang = langCode === "pt" ? "pt-BR" : langCode;
  1356→  1405→
  1357→  1406→    const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT, {
  1358→  1407→      acceptNode(node) {
  1359→  1408→        const parent = node.parentElement;
  1360→  1409→        if (!parent) return NodeFilter.FILTER_REJECT;
  1361→  1410→        if ("SCRIPT STYLE TEXTAREA INPUT".includes(parent.tagName)) return NodeFilter.FILTER_REJECT;
  1362→  1411→        if (hasNoTranslateAncestor(node)) return NodeFilter.FILTER_REJECT;
  1363→  1412→        return NodeFilter.FILTER_ACCEPT;
  1364→  1413→      },
  1365→  1414→    });
  1366→  1415→
  1367→  1416→    const nodes: Text[] = [];
  1368→  1417→    let n: Node | null;
  1369→  1418→    while ((n = walker.nextNode())) nodes.push(n as Text);
  1370→  1419→
  1371→  1420→    const untranslated: string[] = [];
  1372→  1421→
  1373→  1422→    for (const textNode of nodes) {
  1374→  1423→      if (hasNoTranslateAncestor(textNode)) continue;
  1375→  1424→      if (!originalText.has(textNode)) originalText.set(textNode, textNode.nodeValue || "");
  1376→  1425→      const source = originalText.get(textNode) || "";
  1377→  1426→      const trimmed = source.trim();
  1378→  1427→      if (!trimmed) continue;
  1379→  1428→
  1380→  1429→      if (langCode === "pt") {
  1381→  1430→        if (textNode.nodeValue !== source) textNode.nodeValue = source;
  1382→  1431→        continue;
  1383→  1432→      }
  1384→  1433→
  1385→  1434→      const translated = lookup(trimmed, langCode, dictionary);
  1386→  1435→      if (translated) {
  1387→  1436→        const nextValue = source === trimmed ? translated : source.replace(trimmed, translated);
  1388→  1437→        if (textNode.nodeValue !== nextValue) textNode.nodeValue = nextValue;
  1389→  1438→      } else if (trimmed.length > 2 && !/^\d+[\,\d]*$/.test(trimmed) && !/^[#\-+°×→←↑↓]*$/.test(trimmed)) {
  1390→  1439→        untranslated.push(trimmed);
  1391→  1440→      }
  1392→  1441→    }
  1393→  1442→
  1394→  1443→    if (langCode !== "pt") {
  1395→  1444→      document.querySelectorAll("input[placeholder], textarea[placeholder]").forEach((el) => {
  1396→  1445→        const htmlEl = el as HTMLInputElement;
  1397→  1446→        const ph = htmlEl.placeholder;
  1398→  1447→        if (!ph) return;
  1399→  1448→        const tr = lookup(ph, langCode, dictionary);
  1400→  1449→        if (tr) htmlEl.placeholder = tr;
  1401→  1450→      });
  1402→  1451→      document.querySelectorAll("[title]").forEach((el) => {
  1403→  1452→        const htmlEl = el as HTMLElement;
  1404→  1453→        const t = htmlEl.getAttribute("title");
  1405→  1454→        if (!t) return;
  1406→  1455→        const tr = lookup(t, langCode, dictionary);
  1407→  1456→        if (tr) htmlEl.setAttribute("title", tr);
  1408→  1457→      });
  1409→  1458→    }
  1410→  1459→
  1411→  1460→    if (untranslated.length > 0 && langCode !== "pt") {
  1412→  1461→      queueApiTranslation(untranslated, langCode);
  1413→  1462→    }
  1414→  1463→  } finally {
  1415→  1464→    isTranslating = false;
  1416→  1465→  }
  1417→  1466→}

  1419→  1468→function queueApiTranslation(texts: string[], langCode: string) {
  1420→  1469→  if (apiDebounceTimer) clearTimeout(apiDebounceTimer);
  1421→  1470→  apiDebounceTimer = setTimeout(async () => {
  1422→  1471→    const unique = [...new Set(texts)].slice(0, 30);
  1423→  1472→    const toFetch = unique.filter((t) => !apiCache.has(t) && !apiCache.has(stripAccents(t)));
  1424→  1473→    if (toFetch.length === 0) return;
  1425→  1474→    try {
  1426→  1475→      const langMap: Record<string, string> = { en: "en", es: "es", fr: "fr", de: "de", ru: "ru" };
  1427→  1476→      const targetLang = langMap[langCode] || "en";
  1428→  1477→      const res = await fetch("/api/translate", {
  1429→  1478→        method: "POST",
  1430→  1479→        headers: { "Content-Type": "application/json" },
  1431→  1480→        body: JSON.stringify({ texts: toFetch, targetLang }),
  1432→  1481→      });
  1433→  1482→      if (!res.ok) return;
  1434→  1483→      const data = await res.json();
  1435→  1484→      const translations: Record<string, string> = data.translations || {};

  1440→  1489→          if (norm !== key) apiCache.set(norm, val);
  1441→  1490→        }
  1442→  1491→      }
  1443→  1492→      if (Object.keys(translations).length > 0) {
  1444→  1493→        translatePage(langCode);
  1445→  1494→      }
  1446→  1495→    } catch {
  1447→  1496→      // Silently fail
  1448→  1497→    }

  1450→  1499→}
