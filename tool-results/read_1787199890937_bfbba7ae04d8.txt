  1376→/** Pre-built case-insensitive EN map (lazy) */
  1377→let _enLower: Record<string, string> | undefined;
  1378→function getEnLower(): Record<string, string> {
  1379→  if (!_enLower) {
  1380→    _enLower = {};
  1381→    for (const [k, v] of Object.entries(EN)) {
  1382→      const lk = k.toLowerCase();
  1383→      if (!_enLower[lk]) _enLower[lk] = v;
  1384→    }
  1385→  }
  1386→  return _enLower;
  1387→}

  1389→/** Check if text looks like a player name (should never be translated) */
  1390→function looksLikeName(text: string): boolean {
  1391→  const t = text.trim();
  1392→  if (t.length < 2 || t.length > 30) return false;
  1393→  // Contains underscore (e.g. SINGLE_PLAYER)
  1394→  if (/_/.test(t)) return true;
  1395→  // All uppercase 3+ chars (e.g. ADMIN)
  1396→  if (/^[A-Z0-9]{3,}$/.test(t)) return true;
  1397→  return false;
  1398→}

  1400→/** Strip diacritics */
  1401→function stripAccents(str: string): string {
  1402→  return str.normalize("NFD").replace(/[\u0300-\u036f]/g, "");
  1403→}

  1405→function hasNoTranslateAncestor(node: Node): boolean {
  1406→  let el: Node | null = node.parentElement;
  1407→  while (el) {
  1408→    if (el instanceof HTMLElement && (el.hasAttribute("data-no-translate") || el.getAttribute("translate") === "no")) return true;
  1409→    el = el.parentElement;
  1410→  }
  1411→  return false;
  1412→}

  1414→/** Collect all text content that is protected by data-no-translate / translate=no */
  1415→function collectProtectedTexts(): Set<string> {
  1416→  const set = new Set<string>();
  1417→  document.querySelectorAll('[data-no-translate], [translate="no"]').forEach(el => {
  1418→    // Collect direct text children
  1419→    for (const child of el.childNodes) {
  1420→      if (child.nodeType === Node.TEXT_NODE) {
  1421→        const t = child.textContent?.trim();
  1422→        if (t && t.length > 0) set.add(t);
  1423→      }
  1424→    }
  1425→  });
  1426→  return set;
  1427→}

  1429→function lookup(text: string, langCode: string, dictionary: Record<string, string> | undefined): string | undefined {
  1430→  const norm = stripAccents(text);
  1431→  const lower = text.toLowerCase();
  1432→  const el = getEnLower();
  1433→  if (langCode === "en") {
  1434→    return EN[text] || EN[norm] || el[lower] || apiCache.get(text) || apiCache.get(norm);
  1435→  }
  1436→  return dictionary?.[text] || EN[text] || el[lower] || dictionary?.[norm] || EN[norm] || apiCache.get(text) || apiCache.get(norm);
  1437→}

  1439→function translatePage(langCode: string) {
  1440→  if (isTranslating) return;
  1441→  isTranslating = true;
  1442→
  1443→  // Disconnect observer during translation to prevent infinite loop
  1444→  if (observer) observer.disconnect();
  1445→
  1446→  try {
  1447→    currentLang = langCode;
  1448→    const dictionary = dictionaries[langCode];
  1449→    document.documentElement.lang = langCode === "pt" ? "pt-BR" : langCode;
  1450→
  1451→    // Collect all protected texts from data-no-translate / translate=no elements
  1452→    const protectedTexts = collectProtectedTexts();
  1453→
  1454→    const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT, {
  1455→      acceptNode(node) {
  1456→        const parent = node.parentElement;
  1457→        if (!parent) return NodeFilter.FILTER_REJECT;
  1458→        if ("SCRIPT STYLE TEXTAREA INPUT".includes(parent.tagName)) return NodeFilter.FILTER_REJECT;
  1459→        if (hasNoTranslateAncestor(node)) return NodeFilter.FILTER_REJECT;
  1460→        return NodeFilter.FILTER_ACCEPT;
  1461→      },
  1462→    });
  1463→
  1464→    const nodes: Text[] = [];
  1465→    let n: Node | null;
  1466→    while ((n = walker.nextNode())) nodes.push(n as Text);
  1467→
  1468→    const untranslated: string[] = [];
  1469→
  1470→    for (const textNode of nodes) {
  1471→      if (hasNoTranslateAncestor(textNode)) continue;
  1472→      if (!originalText.has(textNode)) originalText.set(textNode, textNode.nodeValue || "");
  1473→      const source = originalText.get(textNode) || "";
  1474→      const trimmed = source.trim();
  1475→      if (!trimmed) continue;
  1476→
  1477→      // Skip protected texts (double safety for player names)
  1478→      if (protectedTexts.has(trimmed)) continue;
  1479→      // Skip texts that look like player names
  1480→      if (looksLikeName(trimmed)) continue;
  1481→
  1482→      if (langCode === "pt") {
  1483→        if (textNode.nodeValue !== source) textNode.nodeValue = source;
  1484→        continue;
  1485→      }
  1486→
  1487→      const translated = lookup(trimmed, langCode, dictionary);
  1488→      if (translated) {
  1489→        const nextValue = source === trimmed ? translated : source.replace(trimmed, translated);
  1490→        if (textNode.nodeValue !== nextValue) textNode.nodeValue = nextValue;
  1491→      } else if (trimmed.length > 2 && !/^\d+[\,\d]*$/.test(trimmed) && !/^[#\-+°×→←↑↓]*$/.test(trimmed) && !looksLikeName(trimmed)) {
  1492→        untranslated.push(trimmed);
  1493→      }
  1494→    }
  1495→
  1496→    if (langCode !== "pt") {
  1497→      document.querySelectorAll("input[placeholder], textarea[placeholder]").forEach((el) => {
  1498→        const htmlEl = el as HTMLInputElement;
  1499→        const ph = htmlEl.placeholder;
  1500→        if (!ph) return;
  1501→        if (protectedTexts.has(ph)) return;
  1502→        const tr = lookup(ph, langCode, dictionary);
  1503→        if (tr) htmlEl.placeholder = tr;
  1504→      });
  1505→      document.querySelectorAll("[title]").forEach((el) => {
  1506→        const htmlEl = el as HTMLElement;
  1507→        const t = htmlEl.getAttribute("title");
  1508→        if (!t) return;
  1509→        if (protectedTexts.has(t)) return;
  1510→        const tr = lookup(t, langCode, dictionary);
  1511→        if (tr) htmlEl.setAttribute("title", tr);
  1512→      });
  1513→    }
  1514→
  1515→    // Filter out protected texts and name-like texts from API queue
  1516→    const filtered = untranslated.filter(t => !protectedTexts.has(t) && !looksLikeName(t));
  1517→    if (filtered.length > 0 && langCode !== "pt") {
  1518→      queueApiTranslation(filtered, langCode);
  1519→    }
  1520→  } finally {
  1521→    isTranslating = false;
  1522→    // Reconnect observer after translation
  1523→    if (observer && currentLang !== "pt") {
  1524→      observer.observe(document.body, { childList: true, subtree: true, characterData: true });
  1525→    }
  1526→  }
  1527→}

  1529→function queueApiTranslation(texts: string[], langCode: string) {
  1530→  if (apiDebounceTimer) clearTimeout(apiDebounceTimer);
  1531→  apiDebounceTimer = setTimeout(async () => {
  1532→    const unique = [...new Set(texts)].slice(0, 30);
  1533→    const toFetch = unique.filter((t) => !apiCache.has(t) && !apiCache.has(stripAccents(t)) && !looksLikeName(t));
  1534→    if (toFetch.length === 0) return;
  1535→    try {
  1536→      const langMap: Record<string, string> = { en: "en", es: "es", fr: "fr", de: "de", ru: "ru" };
  1537→      const targetLang = langMap[langCode] || "en";
  1538→      const res = await fetch("/api/translate", {
  1539→        method: "POST",
  1540→        headers: { "Content-Type": "application/json" },
  1541→        body: JSON.stringify({ texts: toFetch, targetLang }),
  1542→      });
  1543→      if (!res.ok) return;
  1544→      const data = await res.json();
  1545→      const translations: Record<string, string> = data.translations || {};
  1546→      for (const [key, val] of Object.entries(translations)) {
  1547→        if (val && val !== key) {
  1548→          apiCache.set(key, val);
  1549→          const norm = stripAccents(key);
  1550→          if (norm !== key) apiCache.set(norm, val);
  1551→        }
  1552→      }
  1553→      if (Object.keys(translations).length > 0) {
  1554→        translatePage(langCode);
  1555→      }
  1556→    } catch {
  1557→      // Silently fail
  1558→    }
  1559→  }, 400);
  1560→}

  1562→function setLanguage(code: string) {
  1563→  currentLang = code;
  1564→  localStorage.setItem("dayr-language", code);
  1565→  if (code === "pt") {
  1566→    const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT);
  1567→    let n: Node | null;
  1568→    while ((n = walker.nextNode())) {
  1569→      const textNode = n as Text;
  1570→      const orig = originalText.get(textNode);
  1571→      if (orig !== undefined && textNode.nodeValue !== orig) textNode.nodeValue = orig;
  1572→    }
  1573→    document.documentElement.lang = "pt-BR";
  1574→    if (observer) { observer.disconnect(); observer = null; }
  1575→    apiCache.clear();
  1576→    document.querySelectorAll("input[placeholder], textarea[placeholder]").forEach((el) => {
  1577→      const htmlEl = el as HTMLInputElement;
  1578→      const key = htmlEl.getAttribute("data-original-placeholder");
  1579→      if (key) htmlEl.placeholder = key;
  1580→    });
  1581→    return;
  1582→  }
  1583→
  1584→  translatePage(code);
  1585→
  1586→  if (!observer) {
  1587→    let observerDebounce: ReturnType<typeof setTimeout> | null = null;
  1588→    observer = new MutationObserver(() => {
  1589→      if (observerDebounce) clearTimeout(observerDebounce);
  1590→      observerDebounce = setTimeout(() => translatePage(currentLang), 150);
  1591→    });
  1592→    observer.observe(document.body, { childList: true, subtree: true, characterData: true });
  1593→  }
  1594→}
