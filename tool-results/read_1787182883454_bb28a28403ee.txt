  1126→  1126→function translatePage(langCode: string) {
  1127→  1127→  currentLang = langCode;
  1128→  1128→  const dictionary = dictionaries[langCode];
  1129→  1129→  document.documentElement.lang = langCode === "pt" ? "pt-BR" : langCode;
  1130→  1130→
  1131→  1131→  const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT, {
  1132→  1132→    acceptNode(node) {
  1133→  1133→      const parent = node.parentElement;
  1134→  1134→      if (!parent) return NodeFilter.FILTER_REJECT;
  1135→  1135→      if ("SCRIPT STYLE TEXTAREA INPUT".includes(parent.tagName)) return NodeFilter.FILTER_REJECT;
  1136→  1136→      if (hasNoTranslateAncestor(node)) return NodeFilter.FILTER_REJECT;
  1137→  1137→      return NodeFilter.FILTER_ACCEPT;
  1138→  1138→    },
  1139→  1139→  });
  1140→  1140→
  1141→  1141→  const nodes: Text[] = [];
  1142→  1142→  let node: Node | null;
  1143→  1143→  while ((node = walker.nextNode())) nodes.push(node as Text);
  1144→  1144→
  1145→  1145→  const untranslated: string[] = [];
  1146→  1146→
  1147→  1147→  for (const textNode of nodes) {
  1148→  1148→    if (!originalText.has(textNode)) originalText.set(textNode, textNode.nodeValue || "");
  1149→  1149→    const source = originalText.get(textNode) || "";
  1150→  1150→    const trimmed = source.trim();
  1151→  1151→    if (!trimmed) continue;
  1152→  1152→
  1153→  1153→    if (langCode === "pt") {
  1154→  1154→      if (textNode.nodeValue !== source) textNode.nodeValue = source;
  1155→  1155→      continue;
  1156→  1156→    }
  1157→  1157→
  1158→  1158→    const translated = dictionary?.[trimmed] || (langCode !== "en" ? EN[trimmed] : null) || apiCache.get(trimmed);
  1159→  1159→    if (translated) {
  1160→  1160→      const nextValue = source === trimmed ? translated : source.replace(trimmed, translated);
  1161→  1161→      if (textNode.nodeValue !== nextValue) textNode.nodeValue = nextValue;
  1162→  1162→    } else if (trimmed.length > 2 && !/^\d+[,\d]*$/.test(trimmed) && !/^[#\-+°×→←↑↓]*$/.test(trimmed)) {
  1163→  1163→      untranslated.push(trimmed);
  1164→  1164→    }
  1165→  1165→  }
  1166→  1166→
  1167→  1167→  // Also translate placeholders and titles
  1168→  1168→  if (langCode !== "pt") {
  1169→  1169→    document.querySelectorAll("input[placeholder], textarea[placeholder]").forEach((el) => {
  1170→  1170→      const htmlEl = el as HTMLInputElement;
  1171→  1171→      const ph = htmlEl.placeholder;
  1172→  1172→      if (!ph) return;
  1173→  1173→      const tr = dictionary?.[ph] || (langCode !== "en" ? EN[ph] : null) || apiCache.get(ph);
  1174→  1174→      if (tr) htmlEl.placeholder = tr;
  1175→  1175→    });
  1176→  1176→    document.querySelectorAll("[title]").forEach((el) => {
  1177→  1177→      const htmlEl = el as HTMLElement;
  1178→  1178→      const t = htmlEl.getAttribute("title");
  1179→  1179→      if (!t) return;
  1180→  1180→      const tr = dictionary?.[t] || (langCode !== "en" ? EN[t] : null) || apiCache.get(t);
  1181→  1181→      if (tr) htmlEl.setAttribute("title", tr);
  1182→  1182→    });
  1183→  1183→  }
  1184→  1184→
  1185→  1185→  // Debounced API fallback
  1186→  1186→  if (untranslated.length > 0 && langCode !== "pt") {
  1187→  1187→    queueApiTranslation(untranslated, langCode);
  1188→  1188→  }
  1189→  1189→}

  1191→  1191→function queueApiTranslation(texts: string[], langCode: string) {
  1192→  1192→  if (apiDebounceTimer) clearTimeout(apiDebounceTimer);
  1193→  1193→  apiDebounceTimer = setTimeout(async () => {
  1194→  1194→    const unique = [...new Set(texts)].slice(0, 30);
  1195→  1195→    const toFetch = unique.filter((t) => !apiCache.has(t));
  1196→  1196→    if (toFetch.length === 0) return;
  1197→  1197→    try {
  1198→  1198→      const langMap: Record<string, string> = { en: "en", es: "es", fr: "fr", de: "de", ru: "ru" };
  1199→  1199→      const targetLang = langMap[langCode] || "en";
  1200→  1200→      const res = await fetch("/api/translate", {
  1201→  1201→        method: "POST",
  1202→  1202→        headers: { "Content-Type": "application/json" },
  1203→  1203→        body: JSON.stringify({ texts: toFetch, targetLang }),
  1204→  1204→      });
  1205→  1205→      if (!res.ok) return;
  1206→  1206→      const data = await res.json();
  1207→  1207→      const translations: Record<string, string> = data.translations || {};
  1208→  1208→      for (const [key, val] of Object.entries(translations)) {
  1209→  1209→        if (val && val !== key) apiCache.set(key, val);
  1210→  1210→      }
  1211→  1211→      // Re-translate with new cache entries
  1212→  1212→      if (Object.keys(translations).length > 0) {
  1213→  1213→        translatePage(langCode);
  1214→  1214→      }
  1215→  1215→    } catch {
  1216→  1216→      // Silently fail - dictionary covers most strings
  1217→  1217→    }
  1218→  1218→  }, 400);
  1219→  1219→}

  1221→  1221→function setLanguage(code: string) {
  1222→  1222→  currentLang = code;
  1223→  1223→  localStorage.setItem("dayr-language", code);
  1224→  1224→  // Reset all text nodes to original before translating
  1225→  1225→  if (code === "pt") {

  1227→  1227→    let node: Node | null;
  1228→  1228→    while ((node = walker.nextNode())) {

  1230→  1230→      const orig = originalText.get(textNode);
  1231→  1231→      if (orig !== undefined && textNode.nodeValue !== orig) textNode.nodeValue = orig;
  1232→  1232→    }

  1234→  1234→    if (observer) { observer.disconnect(); observer = null; }
  1235→  1235→    // Reset placeholders

  1237→  1237→      const htmlEl = el as HTMLInputElement;

  1242→  1242→  }
  1243→  1243→
  1244→  1244→  translatePage(code);
  1245→  1245→
  1246→  1246→  if (!observer) {
  1247→  1247→    observer = new MutationObserver(() => {
  1248→  1248→ translatePage(currentLang);
  1249→  1249→    });
  1250→  1250→    observer.observe(document.body, { childList: true, subtree: true, characterData: true });
  1251→  1251→  }
