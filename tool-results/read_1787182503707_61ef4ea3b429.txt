    80→function Sparkline({ itemId }: { itemId: string }) {
    81→  const trend = priceTrends.trends[itemId];
    82→  if (!trend || trend.history.length < 2) return null;
    83→  const data = trend.history;
    84→  const min = Math.min(...data);
    85→  const max = Math.max(...data);
    86→  const range = max - min || 1;
    87→  const width = 50;
    88→  const height = 18;
    89→  const padding = 2;
    90→  const points = data
    91→    .map((value, index) => {
    92→      const x = (index / (data.length - 1)) * (width - padding * 2) + padding;
    93→      const y = height - ((value - min) / range) * (height - padding * 2) - padding;
    94→      return `${x},${y}`;
    95→    })
    96→    .join(" ");
    97→  const color = trend.trend === "up" ? "#22c55e" : trend.trend === "down" ? "#ef4444" : "#64748b";
    98→  return (
    99→    <div className="flex items-center gap-1">
   100→      <svg width={width} height={height} className="shrink-0">
   101→        <polyline points={points} fill="none" stroke={color} strokeWidth="1.5" vectorEffect="non-scaling-stroke" />
   102→      </svg>
   103→      {trend.change !== 0 && (
   104→        <span className={`text-[10px] font-bold flex items-center gap-0.5 ${trend.trend === "up" ? "text-green-400" : "text-red-400"}`}>
   105→          {trend.trend === "up" ? <TrendingUp className="w-2.5 h-2.5" /> : <TrendingDown className="w-2.5 h-2.5" />}
   106→          {trend.change > 0 ? "+" : ""}{trend.change}%
   107→        </span>
   108→      )}
   109→    </div>
   110→  );
   111→}
