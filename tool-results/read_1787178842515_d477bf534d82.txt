   787→   TRANSLATION ENGINE

   789→const originalText = new WeakMap<Text, string>();
   790→let observer: MutationObserver | null = null;

   792→let apiDebounceTimer: ReturnType<typeof setTimeout> | null = null;
   793→const apiCache = new Map<string, string>();

   795→function hasNoTranslateAncestor(node: Node): boolean {
   796→  let el: Node | null = node.parentElement;
   797→  while (el) {
   798→    if (el instanceof HTMLElement && el.hasAttribute("data-no-translate")) return true;
   799→    el = el.parentElement;
   800→  }
   801→  return false;
   802→}

   804→function translatePage(langCode: string) {
   805→  currentLang = langCode;
   806→  const dictionary = dictionaries[langCode];
   807→  document.documentElement.lang = langCode === "pt" ? "pt-BR" : langCode;
   808→
   809→  const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT, {
   810→    acceptNode(node) {
   811→      const parent = node.parentElement;
   812→      if (!parent) return NodeFilter.FILTER_REJECT;
   813→      if ("SCRIPT STYLE TEXTAREA INPUT".includes(parent.tagName)) return NodeFilter.FILTER_REJECT;
   814→      if (hasNoTranslateAncestor(node)) return NodeFilter.FILTER_REJECT;
   815→      return NodeFilter.FILTER_ACCEPT;
   816→    },
   817→  });
   818→
   819→  const nodes: Text[] = [];
   820→  let node: Node | null;
   821→  while ((node = walker.nextNode())) nodes.push(node as Text);
   822→
   823→  const untranslated: string[] = [];
   824→
   825→  for (const textNode of nodes) {
   826→    if (!originalText.has(textNode)) originalText.set(textNode, textNode.nodeValue || "");
   827→    const source = originalText.get(textNode) || "";
   828→    const trimmed = source.trim();
   829→    if (!trimmed) continue;
   830→
   831→    if (langCode === "pt") {
   832→      if (textNode.nodeValue !== source) textNode.nodeValue = source;
   833→      continue;
   834→    }
   835→
   836→    const translated = dictionary?.[trimmed] || apiCache.get(trimmed);
   837→    if (translated) {
   838→      const nextValue = source === trimmed ? translated : source.replace(trimmed, translated);
   839→      if (textNode.nodeValue !== nextValue) textNode.nodeValue = nextValue;
   840→    } else if (trimmed.length > 2 && !/^\d+[,\d]*$/.test(trimmed) && !/^[#\-+°×→←↑↓]*$/.test(trimmed)) {
   841→      untranslated.push(trimmed);
   842→    }
   843→  }
   844→
   845→  // Also translate placeholders and titles
   846→  if (langCode !== "pt") {
   847→    document.querySelectorAll("input[placeholder], textarea[placeholder]").forEach((el) => {
   848→      const htmlEl = el as HTMLInputElement;
   849→      const ph = htmlEl.placeholder;
   850→      if (!ph) return;
   851→      const tr = dictionary?.[ph] || apiCache.get(ph);
   852→      if (tr) htmlEl.placeholder = tr;
   853→    });
   854→    document.querySelectorAll("[title]").forEach((el) => {
   855→      const htmlEl = el as HTMLElement;
   856→      const t = htmlEl.getAttribute("title");
   857→      if (!t) return;
   858→      const tr = dictionary?.[t] || apiCache.get(t);
   859→      if (tr) htmlEl.setAttribute("title", tr);
   860→    });
   861→  }
   862→
   863→  // Debounced API fallback
   864→  if (untranslated.length > 0 && langCode !== "pt") {
   865→    queueApiTranslation(untranslated, langCode);
   866→  }
   867→}
   868→
   869→function queueApiTranslation(texts: string[], langCode: string) {
   870→  if (apiDebounceTimer) clearTimeout(apiDebounceTimer);
   871→  apiDebounceTimer = setTimeout(async () => {
   872→    const unique = [...new Set(texts)].slice(0, 30);
   873→    const toFetch = unique.filter((t) => !apiCache.has(t));
   874→    if (toFetch.length === 0) return;
   875→    try {
   876→      const langMap: Record<string, string> = { en: "en", es: "es", fr: "fr", de: "de", ru: "ru" };
   877→      const targetLang = langMap[langCode] || "en";
   878→      const res = await fetch("/api/translate", {
   879→        method: "POST",
   880→        headers: { "Content-Type": "application/json" },
   881→        body: JSON.stringify({ texts: toFetch, targetLang }),
   882→      });
   883→      if (!res.ok) return;
   884→      const data = await res.json();
   885→      const translations: Record<string, string> = data.translations || {};
   886→      for (const [key, val] of Object.entries(translations)) {
   887→        if (val && val !== key) apiCache.set(key, val);
   888→      }
   889→      // Re-translate with new cache entries
   890→      if (Object.keys(translations).length > 0) {
   891→        translatePage(langCode);
   892→      }
   893→    } catch {
   894→      // Silently fail - dictionary covers most strings
   895→    }
   896→  }, 400);
   897→}
   898→
   899→function setLanguage(code: string) {
   900→  currentLang = code;
   901→  localStorage.setItem("dayr-language", code);
   902→  // Reset all text nodes to original before translating
   903→  if (code === "pt") {
   904→    const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT);
   905→    let node: Node | null;
   906→    while ((node = walker.nextNode())) {
   907→      const textNode = node as Text;
   908→      const orig = originalText.get(textNode);
   909→      if (orig !== undefined && textNode.nodeValue !== orig) textNode.nodeValue = orig;
   910→    }
   911→    document.documentElement.lang = "pt-BR";
   912→    if (observer) { observer.disconnect(); observer = null; }
   913→    // Reset placeholders
   914→    document.querySelectorAll("input[placeholder], textarea[placeholder]").forEach((el) => {
   915→      const htmlEl = el as HTMLInputElement;
   916→      const key = htmlEl.getAttribute("data-original-placeholder");
   917→      if (key) htmlEl.placeholder = key;
   918→    });
   919→    return;
   920→  }
   921→
   922→  translatePage(code);
   923→
   924→  if (!observer) {
   925→    observer = new MutationObserver(() => {
   926→ translatePage(currentLang);
   927→    });
   928→    observer.observe(document.body, { childList: true, subtree: true, characterData: true });
   929→  }
   930→}
