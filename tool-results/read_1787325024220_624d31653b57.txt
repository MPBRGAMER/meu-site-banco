  1566→async function loadItemIndex() {
  1567→  if (itemLoaded) return;
  1568→  itemLoaded = true;
  1569→  try {
  1570→    const res = await fetch("/items/_pt_index.json");
  1571→    if (!res.ok) return;
  1572→    const data: Array<{ pt: string; file: string }> = await res.json();
  1573→    itemPtToEn = {};
  1574→    for (const item of data) {
  1575→      // Convert snake_case file name to Title Case English
  1576→      const en = item.file
  1577→        .split("_")
  1578→        .map((w, i) => {
  1579→          // Keep leading number groups together (e.g. "7_62x25mm" → "7 62x25mm")
  1580→          if (i === 0 && /^\d/.test(w)) return w.replace(/(\d+)([a-z])/, "$1 $2").replace(/\b\w/g, c => c.toUpperCase());
  1581→          return w.charAt(0).toUpperCase() + w.slice(1);
  1582→        })
  1583→        .join(" ");
  1584→      itemPtToEn[item.pt] = en;
  1585→      const lower = item.pt.toLowerCase();
  1586→      if (lower !== item.pt) itemPtToEn[lower] = en;
  1587→      const stripped = stripAccents(lower);
  1588→      if (stripped !== lower) itemPtToEn[stripped] = en;
  1589→      // Also store the title-cased version
  1590→      const title = item.pt.replace(/\b\w/g, c => c.toUpperCase());
  1591→      if (title !== item.pt) itemPtToEn[title] = en;
  1592→    }
  1593→  } catch { /* silent */ }
  1594→}

  1598→/** Pre-built case-insensitive EN map (lazy) */

  1600→function getEnLower(): Record<string, string> {
  1601→  if (!_enLower) {
  1602→    _enLower = {};
  1603→    for (const [k, v] of Object.entries(EN)) {

  1607→      const sk = stripAccents(lk);

  1614→/** Check if text looks like a player name (should never be translated) */
  1615→function looksLikeName(text: string): boolean {
  1616→  const t = text.trim();
  1617→  if (t.length < 2 || t.length > 30) return false;
  1618→  if (t === "Admin" || t === "Membro") return false; // real UI words
  1619→  // Contains underscore (e.g. SINGLE_PLAYER)
  1620→  if (/_/.test(t)) return true;
  1621→  // All uppercase 3+ chars (e.g. ADMIN)
  1622→  if (/^[A-Z0-9]{3,}$/.test(t)) return true;
  1623→  // Mixed case without spaces – camelCase or PascalCase nicks (e.g. MPBRGAMER, RedFoot)
  1624→  if (!/\s/.test(t) && /[a-z]/.test(t) && /[A-Z]/.test(t) && !/^[A-ZÁÉÍÓÚÂÊÎÔÛÃÕÇ][a-záéíóúâêîôûãõç]+/.test(t)) return true;
  1625→  // Digits mixed with letters (e.g. Player123)
  1626→  if (/\d/.test(t) && /[a-zA-Z]/.test(t) && t.length <= 20) return true;
  1627→  // Parenthesized server tags (e.g. "(NES) FERNAL30")
  1628→  if (/^\([A-Z]+\)/.test(t)) return true;
  1629→  return false;
  1630→}

  1637→function hasNoTranslateAncestor(node: Node): boolean {
  1638→  let el: Node | null = node.parentElement;
  1639→  while (el) {
  1640→    if (el instanceof HTMLElement && (el.hasAttribute("data-no-translate") || el.getAttribute("translate") === "no")) return true;
  1641→    el = el.parentElement;
  1642→  }
  1643→  return false;
  1644→}

  1646→/** Collect all text content that is protected by data-no-translate / translate=no */
  1647→function collectProtectedTexts(): Set<string> {
  1648→  const set = new Set<string>();
  1649→  document.querySelectorAll('[data-no-translate], [translate="no"]').forEach(el => {
  1650→    // Use TreeWalker to collect ALL text nodes within protected elements (not just direct children)
  1651→    const walker = document.createTreeWalker(el, NodeFilter.SHOW_TEXT);
  1652→    let node: Node | null;
  1653→    while ((node = walker.nextNode())) {
  1654→      const t = node.textContent?.trim();
  1655→      if (t && t.length > 0) set.add(t);
  1656→    }
  1657→  });
  1658→  return set;
  1659→}

  1661→function lookup(text: string, langCode: string, dictionary: Record<string, string> | undefined): string | undefined {
  1662→  const norm = stripAccents(text);
  1663→  const lower = text.toLowerCase();
  1664→  const el = getEnLower();
  1665→  // Check dynamic item index first (covers 300+ game items)
  1666→  if (itemPtToEn) {
  1667→    const itemEn = itemPtToEn[text] || itemPtToEn[lower] || itemPtToEn[norm];
  1668→    if (itemEn) return itemEn;
  1669→  }
  1670→  if (langCode === "en") {
  1671→    return EN[text] || EN[norm] || el[lower] || apiCache.get(text) || apiCache.get(norm);
  1672→  }
  1673→  return dictionary?.[text] || EN[text] || el[lower] || dictionary?.[norm] || EN[norm] || apiCache.get(text) || apiCache.get(norm);
  1674→}

  1676→function translatePage(langCode: string) {
  1677→  if (isTranslating) return;
  1678→  isTranslating = true;
  1679→
  1680→  // Disconnect observer during translation to prevent infinite loop
  1681→  if (observer) observer.disconnect();
  1682→
  1683→  try {
  1684→    currentLang = langCode;
  1685→    const dictionary = dictionaries[langCode];
  1686→    document.documentElement.lang = langCode === "pt" ? "pt-BR" : langCode;
  1687→
  1688→    // Collect all protected texts from data-no-translate / translate=no elements
  1689→    const protectedTexts = collectProtectedTexts();
  1690→
  1691→    const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT, {
  1692→      acceptNode(node) {
  1693→        const parent = node.parentElement;
  1694→        if (!parent) return NodeFilter.FILTER_REJECT;
  1695→        if ("SCRIPT STYLE TEXTAREA INPUT".includes(parent.tagName)) return NodeFilter.FILTER_REJECT;
  1696→        if (hasNoTranslateAncestor(node)) return NodeFilter.FILTER_REJECT;
  1697→        return NodeFilter.FILTER_ACCEPT;
  1698→      },
  1699→    });
  1700→
  1701→    const nodes: Text[] = [];
  1702→    let n: Node | null;
  1703→    while ((n = walker.nextNode())) nodes.push(n as Text);
  1704→
  1705→    const untranslated: string[] = [];
  1706→
  1707→    for (const textNode of nodes) {
  1708→      if (hasNoTranslateAncestor(textNode)) continue;
  1709→      if (!originalText.has(textNode)) originalText.set(textNode, textNode.nodeValue || "");
  1710→      const source = originalText.get(textNode) || "";
  1711→      const trimmed = source.trim();
  1712→      if (!trimmed) continue;
  1713→
  1714→      // Skip protected texts (double safety for player names)
  1715→      if (protectedTexts.has(trimmed)) continue;
  1716→      // Skip texts that look like player names
  1717→      if (looksLikeName(trimmed)) continue;
  1718→
  1719→      if (langCode === "pt") {
  1720→        if (textNode.nodeValue !== source) textNode.nodeValue = source;
  1721→        continue;
  1722→      }
  1723→
  1724→      const translated = lookup(trimmed, langCode, dictionary);
  1725→      if (translated) {
  1726→        const nextValue = source === trimmed ? translated : source.replace(trimmed, translated);
  1727→        if (textNode.nodeValue !== nextValue) textNode.nodeValue = nextValue;
  1728→      } else if (trimmed.length > 2 && !/^\d+[\,\d]*$/.test(trimmed) && !/^[#\-+°×→←↑↓]*$/.test(trimmed) && !looksLikeName(trimmed)) {
  1729→        untranslated.push(trimmed);
  1730→      }
  1731→    }
  1732→
  1733→    if (langCode !== "pt") {
  1734→      document.querySelectorAll("input[placeholder], textarea[placeholder]").forEach((el) => {
  1735→        const htmlEl = el as HTMLInputElement;
  1736→        const ph = htmlEl.placeholder;
  1737→        if (!ph) return;
  1738→        if (protectedTexts.has(ph)) return;
  1739→        const tr = lookup(ph, langCode, dictionary);
  1740→        if (tr) htmlEl.placeholder = tr;
  1741→      });
  1742→      document.querySelectorAll("[title]").forEach((el) => {
  1743→        const htmlEl = el as HTMLElement;
  1744→        const t = htmlEl.getAttribute("title");
  1745→        if (!t) return;
  1746→        if (protectedTexts.has(t)) return;
  1747→        const tr = lookup(t, langCode, dictionary);
  1748→        if (tr) htmlEl.setAttribute("title", tr);
  1749→      });
  1750→    }
  1751→
  1752→    // Filter out protected texts and name-like texts from API queue
  1753→    const filtered = untranslated.filter(t => !protectedTexts.has(t) && !looksLikeName(t));
  1754→    if (filtered.length > 0 && langCode !== "pt") {
  1755→      queueApiTranslation(filtered, langCode);
  1756→    }
  1757→  } finally {
  1758→    isTranslating = false;
  1759→    // Reconnect observer after translation
  1760→    if (observer && currentLang !== "pt") {
  1761→      observer.observe(document.body, { childList: true, subtree: true, characterData: true });
  1762→    }
  1763→  }
  1764→}

  1766→function queueApiTranslation(texts: string[], langCode: string) {
  1767→  if (apiDebounceTimer) clearTimeout(apiDebounceTimer);
  1768→  apiDebounceTimer = setTimeout(async () => {
  1769→    const unique = [...new Set(texts)].slice(0, 30);
  1770→    const toFetch = unique.filter((t) => !apiCache.has(t) && !apiCache.has(stripAccents(t)) && !looksLikeName(t));
  1771→    if (toFetch.length === 0) return;
  1772→    try {
  1773→      const langMap: Record<string, string> = { en: "en", es: "es", fr: "fr", de: "de", ru: "ru" };
  1774→      const targetLang = langMap[langCode] || "en";
  1775→      const res = await fetch("/api/translate", {
  1776→        method: "POST",
  1777→        headers: { "Content-Type": "application/json" },
  1778→        body: JSON.stringify({ texts: toFetch, targetLang }),
  1779→      });
  1780→      if (!res.ok) return;
  1781→      const data = await res.json();
  1782→      const translations: Record<string, string> = data.translations || {};
  1783→      for (const [key, val] of Object.entries(translations)) {
  1784→        if (val && val !== key) {
  1785→          apiCache.set(key, val);
  1786→          const norm = stripAccents(key);
  1787→          if (norm !== key) apiCache.set(norm, val);
  1788→        }
  1789→      }
  1790→      if (Object.keys(translations).length > 0) {
  1791→        translatePage(langCode);
  1792→      }
  1793→    } catch {
  1794→      // Silently fail
  1795→    }
  1796→  }, 400);
  1797→}

  1799→function setLanguage(code: string) {
  1800→  currentLang = code;
  1801→  localStorage.setItem("dayr-language", code);
  1802→  if (code === "pt") {
  1803→    const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT);
  1804→    let n: Node | null;
  1805→    while ((n = walker.nextNode())) {
  1806→      const textNode = n as Text;
  1807→      const orig = originalText.get(textNode);
  1808→      if (orig !== undefined && textNode.nodeValue !== orig) textNode.nodeValue = orig;
  1809→    }
  1810→    document.documentElement.lang = "pt-BR";
  1811→    if (observer) { observer.disconnect(); observer = null; }
  1812→    apiCache.clear();
  1813→    document.querySelectorAll("input[placeholder], textarea[placeholder]").forEach((el) => {
  1814→      const htmlEl = el as HTMLInputElement;
  1815→      const key = htmlEl.getAttribute("data-original-placeholder");
  1816→      if (key) htmlEl.placeholder = key;
  1817→    });
  1818→    return;
  1819→  }
  1820→
  1821→  translatePage(code);
  1822→
  1823→  if (!observer) {
  1824→    let observerDebounce: ReturnType<typeof setTimeout> | null = null;
  1825→    observer = new MutationObserver(() => {
  1826→      if (observerDebounce) clearTimeout(observerDebounce);
  1827→      observerDebounce = setTimeout(() => translatePage(currentLang), 150);
  1828→    });
  1829→    observer.observe(document.body, { childList: true, subtree: true, characterData: true });
  1830→  }
  1831→}
