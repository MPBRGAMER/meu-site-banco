   175→   175→function dateStr(d: string | Date): string {
   176→   176→  if (!d) return "";
   177→   177→  return d instanceof Date ? d.toISOString() : d;
   178→   178→}
   179→   179→
   180→   180→function parseDates<T extends Record<string, unknown>>(
   181→   181→  items: T[],
   182→   182→  dateFields: string[]
   183→   183→): T[] {
   184→   184→  return items.map((item) => {
   185→   185→    const parsed = { ...item };
   186→   186→    for (const f of dateFields) {
   187→   187→      if (parsed[f]) {
   188→   188→        parsed[f] = dateStr(parsed[f] as string) as never;
   189→   189→      }
   190→   190→    }
   191→   191→    return parsed;
   192→   192→  });
   193→   193→}

   195→   195→// === API HELPER ===
   196→   196→async function apiGet(action: string, params?: Record<string, string>): Promise<unknown> {
   197→   197→  const url = new URL("/api/banco", window.location.origin);
   198→   198→  url.searchParams.set("action", action);
   199→   199→  if (params) {
   200→   200→    for (const [k, v] of Object.entries(params)) {
   201→   201→      url.searchParams.set(k, v);
   202→   202→    }
   203→   203→  }
   204→   204→  const res = await fetch(url.toString());
   205→   205→  const data = await res.json();
   206→   206→  if ("error" in data) throw new Error(data.error as string);
   207→   207→  return data;
   208→   208→}
   209→   209→
   210→   210→async function apiPost(action: string, body: Record<string, unknown>): Promise<unknown> {
   211→   211→  const adminPassword = typeof window !== "undefined" ? sessionStorage.getItem("adminPwd") : null;
   212→   212→  const headers: Record<string, string> = { "Content-Type": "application/json" };
   213→   213→  if (adminPassword) headers["x-admin-password"] = adminPassword;
   214→   214→  const res = await fetch("/api/banco", {
   215→   215→    method: "POST",
   216→   216→    headers,
   217→   217→    body: JSON.stringify({ action, ...body }),
   218→   218→  });
   219→   219→  const data = await res.json();
   220→   220→  if ("error" in data) throw new Error(data.error as string);
   221→   221→  return data;
   222→   222→}
