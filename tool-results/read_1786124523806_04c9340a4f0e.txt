     1→import { db } from "@/lib/db";
     2→import { NextRequest, NextResponse } from "next/server";
     3→
     4→function json(data: unknown, status = 200) {
     5→  return NextResponse.json(data, { status });
     6→}
     7→
     8→function err(msg: string, status = 400) {
     9→  return NextResponse.json({ error: msg }, { status });
    10→}
    11→
    12→const ADMIN_PASSWORD = process.env.ADMIN_PASSWORD || "";
    13→
    14→export async function GET(req: NextRequest) {
    15→  const { searchParams } = req.nextUrl;
    16→  const action = searchParams.get("action");
    17→
    18→  try {
    19→    switch (action) {
    20→      case "verifyAdmin": {
    21→        const pwd = searchParams.get("password");
    22→        if (!pwd) return err("Senha obrigatória", 401);
    23→        if (pwd !== ADMIN_PASSWORD) return err("Senha incorreta", 403);
    24→        return json({ success: true });
    25→      }
    26→      case "listEmprestimos": {
    27→        const data = await db.emprestimo.findMany({
    28→          orderBy: { dataEmprestimo: "desc" },
    29→        });
    30→        return json(data);
    31→      }
    32→      case "listInvestidores": {
    33→        const data = await db.investidor.findMany({
    34→          where: { status: "ativo" },
    35→          orderBy: { ordem: "desc" },
    36→        });
    37→        return json(data);
    38→      }
    39→      case "listTabelasTroca": {
    40→        const data = await db.tabelaTroca.findMany();
    41→        return json(data);
    42→      }
    43→      case "listTrocas": {
    44→        const data = await db.trocaRegistro.findMany({
    45→          orderBy: { data: "desc" },
    46→        });
    47→        return json(data);
    48→      }
    49→      case "listComprasVendas": {
    50→        const data = await db.compraVenda.findMany({
    51→          orderBy: { data: "desc" },
    52→        });
    53→        return json(data);
    54→      }
    55→      case "listCaixa": {
    56→        const data = await db.caixaRegistro.findMany({
    57→          orderBy: { data: "desc" },
    58→        });
    59→        return json(data);
    60→      }
    61→      case "listDoadores": {
    62→        const data = await db.doador.findMany({
    63→          orderBy: { ordem: "desc" },
    64→        });
    65→        return json(data);
    66→      }
    67→      case "listLeiloes": {
    68→        const data = await db.leilao.findMany({
    69→          orderBy: { dataCriacao: "desc" },
    70→        });
    71→        return json(data);
    72→      }
    73→      case "listLances": {
    74→        const leilaoId = searchParams.get("leilaoId");
    75→        if (!leilaoId) return err("leilaoId obrigatório");
    76→        const data = await db.lance.findMany({
    77→          where: { leilaoId },
    78→          orderBy: { valor: "desc" },
    79→        });
    80→        return json(data);
    81→      }
    82→      case "listAllLances": {
    83→        const data = await db.lance.findMany({
    84→          orderBy: { valor: "desc" },
    85→        });
    86→        return json(data);
    87→      }
    88→      case "listSorteios": {
    89→        const data = await db.sorteio.findMany({
    90→          orderBy: { dataCriacao: "desc" },
    91→        });
    92→        return json(data);
    93→      }
    94→      case "getHistoricoSorteios": {
    95→        const sorteios = await db.sorteio.findMany({
    96→          where: { status: "finalizado" },
    97→          orderBy: { dataCriacao: "desc" },
    98→        });
    99→        const historico = await Promise.all(
   100→          sorteios.map(async (s) => {
   101→            const count = await db.participanteSorteio.count({
   102→              where: { sorteioId: s.id },
   103→            });
   104→            return { ...s, totalParticipantes: count };
   105→          })
   106→        );
   107→        return json(historico);
   108→      }
   109→      case "listParticipantes": {
   110→        const sorteioId = searchParams.get("sorteioId");
   111→        if (!sorteioId) return err("sorteioId obrigatório");
   112→        const data = await db.participanteSorteio.findMany({
   113→          where: { sorteioId },
   114→          orderBy: { data: "desc" },
   115→        });
   116→        return json(data);
   117→      }
   118→      case "getLoterica": {
   119→        const data = await db.loterica.findMany({
   120→          where: { status: { in: ["vendas_abertas", "sorteio_realizado"] } },
   121→          orderBy: { dataCriacao: "desc" },
   122→          take: 1,
   123→        });
   124→        return json(data);
   125→      }
   126→      case "getAllLoterica": {
   127→        const data = await db.loterica.findMany({
   128→          orderBy: { dataCriacao: "desc" },
   129→        });
   130→        const historico = await Promise.all(
   131→          data.map(async (l) => {
   132→            const numsVendidos = await db.numeroLoterica.count({
   133→              where: { lotericaId: l.id, comprador: { not: null } },
   134→            });
   135→            return { ...l, totalVendidos: numsVendidos };
   136→          })
   137→        );
   138→        return json(historico);
   139→      }
   140→      case "getNumerosLoterica": {
   141→        const lotericaId = searchParams.get("lotericaId");
   142→        if (!lotericaId) return err("lotericaId obrigatório");
   143→        const data = await db.numeroLoterica.findMany({
   144→          where: { lotericaId },
   145→          orderBy: { numero: "asc" },
   146→        });
   147→        return json(data);
   148→      }
   149→      case "getPriceReports": {
   150→        const itemId = searchParams.get("itemId");
   151→        const where = itemId ? { itemId } : {};
   152→        const data = await db.priceReport.findMany({
   153→          where,
   154→          orderBy: { data: "desc" },
   155→          take: 200,
   156→        });
   157→        return json(data);
   158→      }
   159→      case "getReporterRanking": {
   160→        const reports = await db.priceReport.findMany({
   161→          orderBy: { data: "desc" },
   162→        });
   163→        const grouped: Record<string, { nickname: string; count: number; lastReport: string }> = {};
   164→        for (const r of reports) {
   165→          const key = r.nickname.toLowerCase();
   166→          if (!grouped[key]) {
   167→            grouped[key] = { nickname: r.nickname, count: 0, lastReport: r.data.toISOString() };
   168→          }
   169→          grouped[key].count++;
   170→          if (r.data > new Date(grouped[key].lastReport)) {
   171→            grouped[key].lastReport = r.data.toISOString();
   172→          }
   173→        }
   174→        const ranking = Object.values(grouped)
   175→          .sort((a, b) => b.count - a.count)
   176→          .slice(0, 10);
   177→        return json(ranking);
   178→      }
   179→      default:
   180→        return err("Ação GET desconhecida: " + action);
   181→    }
   182→  } catch (e: unknown) {
   183→    const msg = e instanceof Error ? e.message : "Erro interno";
   184→    return err(msg, 500);
   185→  }
   186→}
   187→
   188→export async function POST(req: NextRequest) {
   189→  try {
   190→    const body = await req.json();
   191→    const { action, ...data } = body;
   192→
   193→    if (!action) return err("Campo 'action' é obrigatório");
   194→
   195→    switch (action) {
   196→      // === EMPRÉSTIMOS ===
   197→      case "addEmprestimo": {
   198→        const { player, item, quantidade, dataEmprestimo, tipoMembro, status } = data;
   199→        if (!player || !item || !quantidade) return err("Campos obrigatórios: player, item, quantidade");
   200→        const emp = await db.emprestimo.create({
   201→          data: {
   202→            player,
   203→            item,
   204→            quantidade: Number(quantidade),
   205→            dataEmprestimo: dataEmprestimo ? new Date(dataEmprestimo) : new Date(),
   206→            tipoMembro: tipoMembro || "comum",
   207→            status: status || "pendente",
   208→          },
   209→        });
   210→        return json(emp);
   211→      }
   212→      case "updateEmprestimo": {
   213→        const { id, status, dataPagamento, itemPagamento, quantidadePaga } = data;
   214→        if (!id) return err("id obrigatório");
   215→        const updateData: Record<string, unknown> = {};
   216→        if (status) updateData.status = status;
   217→        if (dataPagamento) updateData.dataPagamento = new Date(dataPagamento);
   218→        if (itemPagamento) updateData.itemPagamento = itemPagamento;
   219→        if (quantidadePaga !== undefined) updateData.quantidadePaga = Number(quantidadePaga);
   220→        const emp = await db.emprestimo.update({
   221→          where: { id },
   222→          data: updateData,
   223→        });
   224→        return json(emp);
   225→      }
   226→
   227→      // === INVESTIDORES ===
   228→      case "addInvestidor": {
   229→        const { nome, observacao } = data;
   230→        if (!nome) return err("nome obrigatório");
   231→        const inv = await db.investidor.create({
   232→          data: { nome, observacao: observacao || null, ordem: 0 },
   233→        });
   234→        return json(inv);
   235→      }
   236→      case "removeInvestidor": {
   237→        const { id } = data;
   238→        if (!id) return err("id obrigatório");
   239→        await db.investidor.delete({ where: { id } });
   240→        return json({ success: true });
   241→      }
   242→      case "reorderInvestidores": {
   243→        const { updates } = data;
   244→        if (!Array.isArray(updates)) return err("updates deve ser array");
   245→        for (const u of updates) {
   246→          await db.investidor.update({
   247→            where: { id: u.id },
   248→            data: { ordem: u.ordem },
   249→          });
   250→        }
   251→        return json({ success: true });
   252→      }
   253→
   254→      // === TABELAS DE TROCA ===
   255→      case "addTabelaTroca": {
   256→        const { itemBase, quantidadeBase, itemResultado, quantidadeResultado, categoria } = data;
   257→        if (!itemBase || !quantidadeBase || !itemResultado || !quantidadeResultado) {
   258→          return err("Campos obrigatórios: itemBase, quantidadeBase, itemResultado, quantidadeResultado");
   259→        }
   260→        const tab = await db.tabelaTroca.create({
   261→          data: {
   262→            itemBase,
   263→            quantidadeBase: Number(quantidadeBase),
   264→            itemResultado,
   265→            quantidadeResultado: Number(quantidadeResultado),
   266→            categoria: categoria || null,
   267→          },
   268→        });
   269→        return json(tab);
   270→      }
   271→      case "removeTabelaTroca": {
   272→        const { id } = data;
   273→        if (!id) return err("id obrigatório");
   274→        await db.tabelaTroca.delete({ where: { id } });
   275→        return json({ success: true });
   276→      }
   277→
   278→      // === TROCAS ===
   279→      case "addTroca": {
   280→        const { player, itemEnviado, quantidadeEnviada, itemRecebido, quantidadeRecebida, tipoMembro, taxaAplicada, lucroBanco } = data;
   281→        if (!player || !itemEnviado || !quantidadeEnviada || !itemRecebido || !quantidadeRecebida) {
   282→          return err("Campos obrigatórios: player, itemEnviado, quantidadeEnviada, itemRecebido, quantidadeRecebida");
   283→        }
   284→        const troca = await db.trocaRegistro.create({
   285→          data: {
   286→            player,
   287→            itemEnviado,
   288→            quantidadeEnviada: Number(quantidadeEnviada),
   289→            itemRecebido,
   290→            quantidadeRecebida: Number(quantidadeRecebida),
   291→            tipoMembro: tipoMembro || "comum",
   292→            taxaAplicada: Number(taxaAplicada) || 0,
   293→            lucroBanco: Number(lucroBanco) || 0,
   294→          },
   295→        });
   296→        return json(troca);
   297→      }
   298→
   299→      // === COMPRAS E VENDAS ===
   300→      case "addCompraVenda": {
   301→        const { tipo, player, item, quantidade, itemPagamento, valor, observacao } = data;
   302→        if (!tipo || !player || !item || !quantidade || valor === undefined) {
   303→          return err("Campos obrigatórios: tipo, player, item, quantidade, valor");
   304→        }
   305→        const cv = await db.compraVenda.create({
   306→          data: {
   307→            tipo,
   308→            player,
   309→            item,
   310→            quantidade: Number(quantidade),
   311→            itemPagamento: itemPagamento || null,
   312→            valor: Number(valor),
   313→            observacao: observacao || null,
   314→          },
   315→        });
   316→        return json(cv);
   317→      }
   318→
   319→      // === CAIXA ===
   320→      case "addCaixa": {
   321→        const { tipo, descricao, item, quantidade, valor, origem } = data;
   322→        if (!tipo || !descricao || !item || !quantidade || !origem) {
   323→          return err("Campos obrigatórios: tipo, descricao, item, quantidade, origem");
   324→        }
   325→        const reg = await db.caixaRegistro.create({
   326→          data: {
   327→            tipo,
   328→            descricao,
   329→            item,
   330→            quantidade: Number(quantidade),
   331→            valor: valor !== undefined ? Number(valor) : null,
   332→            origem,
   333→          },
   334→        });
   335→        return json(reg);
   336→      }
   337→      case "resetCaixa": {
   338→        await db.caixaRegistro.deleteMany();
   339→        await db.emprestimo.deleteMany();
   340→        await db.trocaRegistro.deleteMany();
   341→        await db.compraVenda.deleteMany();
   342→        await db.doador.deleteMany();
   343→        return json({ success: true });
   344→      }
   345→
   346→      // === DOADORES ===
   347→      case "addDoador": {
   348→        const { nome, item, quantidade } = data;
   349→        if (!nome || !item || !quantidade) return err("Campos obrigatórios: nome, item, quantidade");
   350→        const doador = await db.doador.create({
   351→          data: { nome, item, quantidade: Number(quantidade), ordem: 0 },
   352→        });
   353→        return json(doador);
   354→      }
   355→      case "reorderDoadores": {
   356→        const { updates } = data;
   357→        if (!Array.isArray(updates)) return err("updates deve ser array");
   358→        for (const u of updates) {
   359→          await db.doador.update({
   360→            where: { id: u.id },
   361→            data: { ordem: u.ordem },
   362→          });
   363→        }
   364→        return json({ success: true });
   365→      }
   366→
   367→      // === LEILÕES ===
   368→      case "addLeilao": {
   369→        const { donoItem, nomeItem, imagemUrl, quantidade, valorInicial, moedaAceita, taxaCasa, dataExpiracao, tipoOrigem } = data;
   370→        if (!donoItem || !nomeItem || valorInicial === undefined || !moedaAceita || !dataExpiracao) {
   371→          return err("Campos obrigatórios: donoItem, nomeItem, valorInicial, moedaAceita, dataExpiracao");
   372→        }
   373→        const leilao = await db.leilao.create({
   374→          data: {
   375→            donoItem,
   376→            nomeItem,
   377→            imagemUrl: imagemUrl || null,
   378→            quantidade: Number(quantidade) || 1,
   379→            valorInicial: Number(valorInicial),
   380→            moedaAceita,
   381→            taxaCasa: Number(taxaCasa) || 15,
   382→            dataExpiracao: new Date(dataExpiracao),
   383→            tipoOrigem: tipoOrigem || "comum",
   384→          },
   385→        });
   386→        return json(leilao);
   387→      }
   388→      case "updateLeilao": {
   389→        const { id, ...updateData } = data;
   390→        if (!id) return err("id obrigatório");
   391→        const cleaned: Record<string, unknown> = {};
   392→        for (const [k, v] of Object.entries(updateData)) {
   393→          if (v !== undefined && v !== null) {
   394→            if (k === "valorVencedor") cleaned[k] = Number(v);
   395→            else cleaned[k] = v;
   396→          }
   397→        }
   398→        const leilao = await db.leilao.update({ where: { id }, data: cleaned });
   399→        return json(leilao);
   400→      }
   401→      case "removeLeilao": {
   402→        const { id } = data;
   403→        if (!id) return err("id obrigatório");
   404→        await db.leilao.delete({ where: { id } });
   405→        return json({ success: true });
   406→      }
   407→
   408→      // === LANCES ===
   409→      case "addLance": {
   410→        const { leilaoId, jogador, valor } = data;
   411→        if (!leilaoId || !jogador || valor === undefined) {
   412→          return err("Campos obrigatórios: leilaoId, jogador, valor");
   413→        }
   414→        const leilao = await db.leilao.findUnique({ where: { id: leilaoId } });
   415→        if (!leilao) return err("Leilão não encontrado", 404);
   416→        if (leilao.status === "finalizado") return err("Leilão já finalizado");
   417→        // Bloqueia lances em espera quando o tempo +1min já expirou
   418→        if (leilao.status === "espera" && new Date() >= new Date(leilao.dataExpiracao)) {
   419→          return err("O tempo de disputa acabou. Aguardando finalização.");
   420→        }
   421→
   422→        const lances = await db.lance.findMany({
   423→          where: { leilaoId },
   424→          orderBy: { valor: "desc" },
   425→        });
   426→        const maiorLance = lances.length > 0 ? lances[0] : null;
   427→        const valorMinimo = maiorLance ? maiorLance.valor : leilao.valorInicial;
   428→
   429→        if (Number(valor) <= valorMinimo) {
   430→          return err(`O lance deve ser maior que ${valorMinimo} ${leilao.moedaAceita}`);
   431→        }
   432→
   433→        const lance = await db.lance.create({
   434→          data: { leilaoId, jogador, valor: Number(valor) },
   435→        });
   436→
   437→        // Se o tempo original já passou, qualquer lance adiciona +1min e vai para espera
   438→        const now = new Date();
   439→        const originalExpired = now >= new Date(leilao.dataExpiracao);
   440→        if (originalExpired || leilao.status === "espera") {
   441→          const newDeadline = new Date(now.getTime() + 60000);
   442→          await db.leilao.update({
   443→            where: { id: leilaoId },
   444→            data: { dataUltimoLance: new Date(), dataExpiracao: newDeadline, status: "espera" },
   445→          });
   446→        } else {
   447→          await db.leilao.update({
   448→            where: { id: leilaoId },
   449→            data: { dataUltimoLance: new Date() },
   450→          });
   451→        }
   452→
   453→        return json(lance);
   454→      }
   455→
   456→      // === SORTEIOS ===
   457→      case "addSorteio": {
   458→        const { nomeItem, quantidade, duracaoMinutos } = data;
   459→        if (!nomeItem || !quantidade || !duracaoMinutos) {
   460→          return err("Campos obrigatórios: nomeItem, quantidade, duracaoMinutos");
   461→        }
   462→        const dataFim = new Date(Date.now() + Number(duracaoMinutos) * 60 * 1000);
   463→        const sorteio = await db.sorteio.create({
   464→          data: { nomeItem, quantidade: Number(quantidade), duracaoMinutos: Number(duracaoMinutos), dataFim },
   465→        });
   466→        return json(sorteio);
   467→      }
   468→      case "participarSorteio": {
   469→        const { sorteioId, jogador } = data;
   470→        if (!sorteioId || !jogador) return err("Campos obrigatórios: sorteioId, jogador");
   471→        const sorteio = await db.sorteio.findUnique({ where: { id: sorteioId } });
   472→        if (!sorteio) return err("Sorteio não encontrado", 404);
   473→        if (sorteio.status !== "ativo") return err("Sorteio não está ativo");
   474→        if (sorteio.dataFim && new Date(sorteio.dataFim) < new Date()) {
   475→          return err("O tempo do sorteio acabou");
   476→        }
   477→        const existing = await db.participanteSorteio.findFirst({
   478→          where: { sorteioId, jogador },
   479→        });
   480→        if (existing) return err("Você já está participando deste sorteio");
   481→        const p = await db.participanteSorteio.create({ data: { sorteioId, jogador } });
   482→        return json(p);
   483→      }
   484→      case "sortear": {
   485→        const { sorteioId } = data;
   486→        if (!sorteioId) return err("sorteioId obrigatório");
   487→        const participantes = await db.participanteSorteio.findMany({ where: { sorteioId } });
   488→        if (participantes.length === 0) return err("Nenhum participante no sorteio");
   489→        const randomIndex = Math.floor(Math.random() * participantes.length);
   490→        const ganhador = participantes[randomIndex];
   491→        const updated = await db.sorteio.update({
   492→          where: { id: sorteioId },
   493→          data: { status: "finalizado", ganhador: ganhador.jogador, dataFim: new Date() },
   494→        });
   495→        return json({ sucesso: true, ganhador: ganhador.jogador });
   496→      }
   497→      case "removeSorteio": {
   498→        const { id } = data;
   499→        if (!id) return err("id obrigatório");
   500→        await db.sorteio.delete({ where: { id } });
   501→        return json({ success: true });
   502→      }
   503→
   504→      // === LOTÉRICA ===
   505→      case "criarLoterica": {
   506→        const { valorNumero, moedaAceita, premioMinimo, duracaoMinutos } = data;
   507→        if (!valorNumero || !moedaAceita) return err("Campos obrigatórios: valorNumero, moedaAceita");
   508→        // Verificar se ja existe loterica ativa (nao finalizada)
   509→        const ativa = await db.loterica.findFirst({
   510→          where: { status: { in: ["vendas_abertas", "sorteio_realizado"] } },
   511→        });
   512→        if (ativa) return err("Já existe lotérica ativa. Finalize a atual antes de criar uma nova.");
   513→
   514→        const dur = Number(duracaoMinutos) || 60;
   515→        const prem = Number(premioMinimo) || 0;
   516→        const dataFimVendas = new Date(Date.now() + dur * 60 * 1000);
   517→
   518→        // Buscar premio acumulado da ultima loterica sem ganhador
   519→        let acumulado = 0;
   520→        const ultimaSemGanhador = await db.loterica.findFirst({
   521→          where: { ganhador: null, valorPremio: { gt: 0 } },
   522→          orderBy: { dataCriacao: "desc" },
   523→        });
   524→        if (ultimaSemGanhador) {
   525→          acumulado = ultimaSemGanhador.valorPremio || 0;
   526→        }
   527→
   528→        const lot = await db.loterica.create({
   529→          data: {
   530→            valorNumero: Number(valorNumero),
   531→            moedaAceita,
   532→            premioMinimo: prem,
   533→            premioAcumulado: acumulado,
   534→            duracaoMinutos: dur,
   535→            dataFimVendas,
   536→            status: "vendas_abertas",
   537→          },
   538→        });
   539→        // Create 1000 numbers
   540→        const nums = Array.from({ length: 1000 }, (_, i) => ({
   541→          lotericaId: lot.id,
   542→          numero: i + 1,
   543→        }));
   544→        await db.numeroLoterica.createMany({ data: nums });
   545→        return json({ success: true, lotericaId: lot.id, acumulado });
   546→      }
   547→      case "comprarNumero": {
   548→        const { lotericaId, numero, comprador } = data;
   549→        if (!lotericaId || numero === undefined || !comprador) {
   550→          return err("Campos obrigatórios: lotericaId, numero, comprador");
   551→        }
   552→        const lotData = await db.loterica.findUnique({ where: { id: lotericaId } });
   553→        if (!lotData) return err("Lotérica não encontrada", 404);
   554→        if (lotData.status !== "vendas_abertas") return err("Vendas não estão abertas");
   555→        if (lotData.dataFimVendas && new Date(lotData.dataFimVendas) < new Date()) {
   556→          return err("Tempo de vendas encerrado");
   557→        }
   558→        const numEntry = await db.numeroLoterica.findFirst({
   559→          where: { lotericaId, numero: Number(numero) },
   560→        });
   561→        if (!numEntry) return err("Número inválido");
   562→        if (numEntry.comprador) return err("Número já vendido");
   563→
   564→        await db.numeroLoterica.update({
   565→          where: { id: numEntry.id },
   566→          data: { comprador, dataCompra: new Date() },
   567→        });
   568→
   569→        // Arrecadado total aumenta, mas NAO credita no estoque ainda
   570→        const novoArrecadado = (lotData.arrecadadoTotal || 0) + lotData.valorNumero;
   571→        await db.loterica.update({
   572→          where: { id: lotericaId },
   573→          data: { arrecadadoTotal: novoArrecadado },
   574→        });
   575→
   576→        return json({ success: true });
   577→      }
   578→      case "iniciarSorteioLoterica": {
   579→        const { lotericaId } = data;
   580→        if (!lotericaId) return err("lotericaId obrigatório");
   581→        const numeros = await db.numeroLoterica.findMany({ where: { lotericaId } });
   582→        const numerosVendidos = numeros.filter((n) => n.comprador);
   583→        const lotData = await db.loterica.findUnique({ where: { id: lotericaId } });
   584→        if (!lotData) return err("Lotérica não encontrada", 404);
   585→        if (numerosVendidos.length === 0) return err("Nenhum número vendido");
   586→
   587→        const numeroSorteado = Math.floor(Math.random() * 1000) + 1;
   588→        const ganhador = numeros.find((n) => n.numero === numeroSorteado && n.comprador);
   589→
   590→        // Calcular premio: acumulado SUBSTITUI o minimo (nao soma)
   591→        // Ex: minimo 100, acumulado 160 → minimo efetivo = 160
   592→        const effectiveMin = Math.max(lotData.premioMinimo, lotData.premioAcumulado || 0);
   593→        const premio80 = lotData.arrecadadoTotal * 0.8;
   594→        const premioFinal = Math.max(premio80, effectiveMin);
   595→        const taxaBanco = lotData.arrecadadoTotal * 0.2;
   596→
   597→        await db.loterica.update({
   598→          where: { id: lotericaId },
   599→          data: {
   600→            status: "sorteio_realizado",
   601→            numeroSorteado,
   602→            ganhador: ganhador ? ganhador.comprador : null,
   603→            valorPremio: premioFinal,
   604→            dataSorteio: new Date(),
   605→          },
   606→        });
   607→
   608→        // Creditar 20% das vendas no estoque do banco
   609→        if (taxaBanco > 0) {
   610→          await db.caixaRegistro.create({
   611→            data: {
   612→              tipo: "entrada",
   613→              descricao: `Taxa bancária Lotérica (20% das vendas)`,
   614→              item: lotData.moedaAceita,
   615→              quantidade: Math.round(taxaBanco),
   616→              valor: Math.round(taxaBanco),
   617→              origem: "loterica",
   618→            },
   619→          });
   620→        }
   621→
   622→        // Se teve ganhador, registrar saida do premio
   623→        if (ganhador) {
   624→          await db.caixaRegistro.create({
   625→            data: {
   626→              tipo: "saida",
   627→              descricao: `Prêmio Lotérica - Número ${numeroSorteado} (${ganhador.comprador})`,
   628→              item: lotData.moedaAceita,
   629→              quantidade: Math.round(premioFinal),
   630→              valor: Math.round(premioFinal),
   631→              origem: "loterica_premio",
   632→            },
   633→          });
   634→        } else {
   635→          // Ninguem acertou - o premio vira o novo minimo para a proxima
   636→          await db.loterica.update({
   637→            where: { id: lotericaId },
   638→            data: { premioAcumulado: Math.round(premioFinal) },
   639→          });
   640→        }
   641→
   642→        return json({
   643→          success: true,
   644→          numeroSorteado,
   645→          ganhador: ganhador ? ganhador.comprador : null,
   646→          premioFinal: Math.round(premioFinal),
   647→          taxaBanco: Math.round(taxaBanco),
   648→          acumulou: !ganhador,
   649→        });
   650→      }
   651→      case "finalizarLoterica": {
   652→        const { lotericaId } = data;
   653→        if (!lotericaId) return err("lotericaId obrigatório");
   654→        const lotData = await db.loterica.findUnique({ where: { id: lotericaId } });
   655→        if (!lotData) return err("Lotérica não encontrada", 404);
   656→        if (lotData.status !== "sorteio_realizado") return err("Sorteio ainda não foi realizado");
   657→
   658→        // Marca como finalizada - pronto para criar proxima
   659→        await db.loterica.update({
   660→          where: { id: lotericaId },
   661→          data: { status: "finalizada" },
   662→        });
   663→
   664→        return json({ success: true, acumuladoProxima: lotData.premioAcumulado || 0 });
   665→      }
   666→
   667→      // === PRICE REPORTS ===
   668→      case "reportPrice": {
   669→        const { itemId, itemName, nickname, steelQty, steelPrice, cementQty, cementPrice } = data;
   670→        if (!itemId || !itemName || !nickname || steelQty === undefined || cementQty === undefined) {
   671→          return err("Campos obrigatorios: itemId, itemName, nickname, steelQty, cementQty");
   672→        }
   673→        const report = await db.priceReport.create({
   674→          data: {
   675→            itemId,
   676→            itemName,
   677→            nickname,
   678→            steelQty: Number(steelQty),
   679→            steelPrice: Number(steelPrice) || 1,
   680→            cementQty: Number(cementQty),
   681→            cementPrice: Number(cementPrice) || 1,
   682→          },
   683→        });
   684→        return json(report);
   685→      }
   686→
   687→      default:
   688→        return err("Ação POST desconhecida: " + action);
   689→    }
   690→  } catch (e: unknown) {
   691→    const msg = e instanceof Error ? e.message : "Erro interno";
   692→    return err(msg, 500);
   693→  }
   694→}
