  1376→/** Strip diacritics */

  1378→  return str.normalize("NFD").replace(/[\u0300-\u036f]/g, "");

  1381→function hasNoTranslateAncestor(node: Node): boolean {
  1382→  let el: Node | null = node.parentElement;
  1383→  while (el) {
  1384→    if (el instanceof HTMLElement && (el.hasAttribute("data-no-translate") || el.getAttribute("translate") === "no")) return true;
  1385→    el = el.parentElement;
  1386→  }
  1387→  return false;
  1388→}

  1390→function lookup(text: string, langCode: string, dictionary: Record<string, string> | undefined): string | undefined {
  1391→  const norm = stripAccents(text);
  1392→  if (langCode === "en") {
  1393→    return EN[text] || EN[norm] || apiCache.get(text) || apiCache.get(norm);
  1394→  }
  1395→  return dictionary?.[text] || EN[text] || dictionary?.[norm] || EN[norm] || apiCache.get(text) || apiCache.get(norm);
  1396→}

  1398→function translatePage(langCode: string) {
  1399→  if (isTranslating) return;
  1400→  isTranslating = true;
  1401→  try {
  1402→    currentLang = langCode;
  1403→    const dictionary = dictionaries[langCode];
  1404→    document.documentElement.lang = langCode === "pt" ? "pt-BR" : langCode;
  1405→
  1406→    const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT, {
  1407→      acceptNode(node) {
  1408→        const parent = node.parentElement;
  1409→        if (!parent) return NodeFilter.FILTER_REJECT;
  1410→        if ("SCRIPT STYLE TEXTAREA INPUT".includes(parent.tagName)) return NodeFilter.FILTER_REJECT;
  1411→        if (hasNoTranslateAncestor(node)) return NodeFilter.FILTER_REJECT;
  1412→        return NodeFilter.FILTER_ACCEPT;
  1413→      },
  1414→    });
  1415→
  1416→    const nodes: Text[] = [];
  1417→    let n: Node | null;
  1418→    while ((n = walker.nextNode())) nodes.push(n as Text);
  1419→
  1420→    const untranslated: string[] = [];
  1421→
  1422→    for (const textNode of nodes) {
  1423→      if (hasNoTranslateAncestor(textNode)) continue;
  1424→      if (!originalText.has(textNode)) originalText.set(textNode, textNode.nodeValue || "");
  1425→      const source = originalText.get(textNode) || "";
  1426→      const trimmed = source.trim();
  1427→      if (!trimmed) continue;
  1428→
  1429→      if (langCode === "pt") {
  1430→        if (textNode.nodeValue !== source) textNode.nodeValue = source;
  1431→        continue;
  1432→      }
  1433→
  1434→      const translated = lookup(trimmed, langCode, dictionary);
  1435→      if (translated) {
  1436→        const nextValue = source === trimmed ? translated : source.replace(trimmed, translated);
  1437→        if (textNode.nodeValue !== nextValue) textNode.nodeValue = nextValue;
  1438→      } else if (trimmed.length > 2 && !/^\d+[\,\d]*$/.test(trimmed) && !/^[#\-+°×→←↑↓]*$/.test(trimmed)) {
  1439→        untranslated.push(trimmed);
  1440→      }
  1441→    }
  1442→
  1443→    if (langCode !== "pt") {
  1444→      document.querySelectorAll("input[placeholder], textarea[placeholder]").forEach((el) => {
  1445→        const htmlEl = el as HTMLInputElement;
  1446→        const ph = htmlEl.placeholder;
  1447→        if (!ph) return;
  1448→        const tr = lookup(ph, langCode, dictionary);
  1449→        if (tr) htmlEl.placeholder = tr;
  1450→      });
  1451→      document.querySelectorAll("[title]").forEach((el) => {
  1452→        const htmlEl = el as HTMLElement;
  1453→        const t = htmlEl.getAttribute("title");
  1454→        if (!t) return;
  1455→        const tr = lookup(t, langCode, dictionary);
  1456→        if (tr) htmlEl.setAttribute("title", tr);
  1457→      });
  1458→    }
  1459→
  1460→    if (untranslated.length > 0 && langCode !== "pt") {
  1461→      queueApiTranslation(untranslated, langCode);
  1462→    }
  1463→  } finally {
  1464→    isTranslating = false;
  1465→  }
  1466→}

  1468→function queueApiTranslation(texts: string[], langCode: string) {
  1469→  if (apiDebounceTimer) clearTimeout(apiDebounceTimer);
  1470→  apiDebounceTimer = setTimeout(async () => {
  1471→    const unique = [...new Set(texts)].slice(0, 30);
  1472→    const toFetch = unique.filter((t) => !apiCache.has(t) && !apiCache.has(stripAccents(t)));
  1473→    if (toFetch.length === 0) return;
  1474→    try {
  1475→      const langMap: Record<string, string> = { en: "en", es: "es", fr: "fr", de: "de", ru: "ru" };
  1476→      const targetLang = langMap[langCode] || "en";
  1477→      const res = await fetch("/api/translate", {
  1478→        method: "POST",
  1479→        headers: { "Content-Type": "application/json" },
  1480→        body: JSON.stringify({ texts: toFetch, targetLang }),
  1481→      });
  1482→      if (!res.ok) return;
  1483→      const data = await res.json();
  1484→      const translations: Record<string, string> = data.translations || {};
  1485→      for (const [key, val] of Object.entries(translations)) {
  1486→        if (val && val !== key) {
  1487→          apiCache.set(key, val);
  1488→          const norm = stripAccents(key);
  1489→          if (norm !== key) apiCache.set(norm, val);
  1490→        }
  1491→      }
  1492→      if (Object.keys(translations).length > 0) {
  1493→        translatePage(langCode);
  1494→      }
  1495→    } catch {
  1496→      // Silently fail
  1497→    }
  1498→  }, 400);
  1499→}

  1501→function setLanguage(code: string) {
  1502→  currentLang = code;
  1503→  localStorage.setItem("dayr-language", code);
  1504→  if (code === "pt") {
  1505→    const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT);
  1506→    let n: Node | null;
  1507→    while ((n = walker.nextNode())) {
  1508→      const textNode = n as Text;
  1509→      const orig = originalText.get(textNode);
  1510→      if (orig !== undefined && textNode.nodeValue !== orig) textNode.nodeValue = orig;
  1511→    }
  1512→    document.documentElement.lang = "pt-BR";
  1513→    if (observer) { observer.disconnect(); observer = null; }
  1514→    apiCache.clear();
  1515→    document.querySelectorAll("input[placeholder], textarea[placeholder]").forEach((el) => {
  1516→      const htmlEl = el as HTMLInputElement;
  1517→      const key = htmlEl.getAttribute("data-original-placeholder");
  1518→      if (key) htmlEl.placeholder = key;
  1519→    });
  1520→    return;
  1521→  }
  1522→
  1523→  translatePage(code);
  1524→
  1525→  if (!observer) {
  1526→    let observerDebounce: ReturnType<typeof setTimeout> | null = null;
  1527→    observer = new MutationObserver(() => {
  1528→      if (observerDebounce) clearTimeout(observerDebounce);
  1529→      observerDebounce = setTimeout(() => translatePage(currentLang), 150);
  1530→    });
  1531→    observer.observe(document.body, { childList: true, subtree: true, characterData: true });
  1532→  }
  1533→}
