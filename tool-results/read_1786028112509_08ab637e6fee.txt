    64→function parsePriceValue(price: string): number {
    65→  const parts = price.split(":");
    66→  if (parts.length !== 2) return 9999;
    67→  const qty = parseInt(parts[0]);
    68→  const val = parseInt(parts[1]);
    69→  if (val === 0) return 9999;
    70→  return qty / val;
    71→}

    77→function Sparkline({ itemId }: { itemId: string }) {
    78→  const trend = priceTrends.trends[itemId];
    79→  if (!trend || trend.history.length < 2) return null;
    80→  const data = trend.history;
    81→  const min = Math.min(...data);
    82→  const max = Math.max(...data);
    83→  const range = max - min || 1;
    84→  const width = 50;
    85→  const height = 18;
    86→  const padding = 2;
    87→  const points = data
    88→    .map((value, index) => {
    89→      const x = (index / (data.length - 1)) * (width - padding * 2) + padding;
    90→      const y = height - ((value - min) / range) * (height - padding * 2) - padding;
    91→      return `${x},${y}`;
    92→    })
    93→    .join(" ");
    94→  const color = trend.trend === "up" ? "#22c55e" : trend.trend === "down" ? "#ef4444" : "#64748b";
    95→  return (
    96→    <div className="flex items-center gap-1">
    97→      <svg width={width} height={height} className="shrink-0">
    98→        <polyline points={points} fill="none" stroke={color} strokeWidth="1.5" vectorEffect="non-scaling-stroke" />
    99→      </svg>
   100→      {trend.change !== 0 && (
   101→        <span className={`text-[10px] font-bold flex items-center gap-0.5 ${trend.trend === "up" ? "text-green-400" : "text-red-400"}`}>
   102→          {trend.trend === "up" ? <TrendingUp className="w-2.5 h-2.5" /> : <TrendingDown className="w-2.5 h-2.5" />}
   103→          {trend.change > 0 ? "+" : ""}{trend.change}%
   104→        </span>
   105→      )}
   106→    </div>
   107→  );
   108→}

   460→            else if (sortField === "cement") cmp = parsePriceValue(a.cement) - parsePriceValue(b.cement);

   469→  const totalAllItems = categories.reduce((sum, c) => sum + c.items.length, 0);
   470→  const totalMissing = categories.reduce((sum, c) => sum + c.items.filter((i) => isPriceUnknown(i.steel)).length, 0);
   471→
   472→  const toggleSort = (field: "name" | "steel" | "cement") => {

   474→    else { setSortField(field); setSortDir("asc"); }

   481→      else next.add(id);

   506→            <Pencil className="w-3.5 h-3.5" /> Editar
   507→          </button>

   509→            <BookOpen className="w-3.5 h-3.5" /> Guia
   510→          </button>
   511→        </div>
   512→      </div>

   515→        <div className="flex items-center gap-2">

   519→        <div className="flex items-center gap-2">
   520→          <div className="w-7 h-7 rounded-md bg-orange-500/10 flex items-center justify-center"><AlertCircle className="w-3.5 h-3.5 text-orange-400" /></div>
   521→          <div><p className="text-[10px] text-muted-foreground">Sem Preco</p><p className="text-sm font-bold font-mono text-foreground">{totalMissing}</p></div>
   522→        </div>
   523→        <div className="flex items-center gap-2">
   524→          <div className="w-7 h-7 rounded-md bg-blue-500/10 flex items-center justify-center"><BarChart3 className="w-3.5 h-3.5 text-blue-400" /></div>

   566→              <button onClick={() => toggleCategory(cat.id)} className="w-full px-3 py-2.5 border-b border-border bg-primary/5 hover:bg-primary/10 transition-colors text-left">
   567→                <div className="flex items-center gap-2">
   568→                  {expandedCategories.has(cat.id) ? <ChevronDown className="w-3.5 h-3.5 text-primary" /> : <ChevronRight className="w-3.5 h-3.5 text-muted-foreground" />}
   569→                  <span className="text-sm font-bold text-foreground">{cat.name}</span>

   607→                            <span className="text-[9px] text-muted-foreground">-</span>
   608→                          )}
   609→                        </div>
   610→                      </div>

   615→            </div>

   617→        )}

   620→      <div className="rounded-md border border-border bg-card p-3">
   621→        <div className="flex items-center justify-between mb-2">
   622→          <h3 className="text-xs font-bold text-foreground flex items-center gap-2"><Users className="w-3.5 h-3.5 text-primary" /> Ultimos Reports da Comunidade</h3>
   623→        </div>
   624→        {priceReports.length === 0 ? (
   625→          <p className="text-[11px] text-muted-foreground">Nenhum report ainda. Seja o primeiro a reportar precos!</p>
   626→        ) : (
   627→          <div className="space-y-1 max-h-48 overflow-y-auto">
   628→            {priceReports.slice(0, 20).map((r) => (
   629→              <div key={r.id} className="flex items-center justify-between py-1 px-2 rounded border-b border-border/30 text-[11px]">

   640→          </div>

   642→      </div>
   643→
   644→      <ReportarModal isOpen={showReportar} onClose={() => setShowReportar(false)} items={allItems} onReport={handleReport} />
   645→      <EditarModal isOpen={showEditar} onClose={() => setShowEditar(false)} items={allItems} />
   646→      <GuiaModal isOpen={showGuia} onClose={() => setShowGuia(false)} />
