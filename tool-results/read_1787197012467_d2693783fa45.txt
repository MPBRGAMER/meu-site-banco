  1181→/** Strip diacritics */
  1182→function stripAccents(str: string): string {
  1183→  return str.normalize("NFD").replace(/[\u0300-\u036f]/g, "");

  1186→function hasNoTranslateAncestor(node: Node): boolean {
  1187→  let el: Node | null = node.parentElement;
  1188→  while (el) {
  1189→    if (el instanceof HTMLElement && (el.hasAttribute("data-no-translate") || el.getAttribute("translate") === "no")) return true;
  1190→    el = el.parentElement;
  1191→  }
  1192→  return false;
  1193→}

  1195→function lookup(text: string, langCode: string, dictionary: Record<string, string> | undefined): string | undefined {
  1196→  const norm = stripAccents(text);
  1197→  if (langCode === "en") {
  1198→    return EN[text] || EN[norm] || apiCache.get(text) || apiCache.get(norm);
  1199→  }
  1200→  return dictionary?.[text] || EN[text] || dictionary?.[norm] || EN[norm] || apiCache.get(text) || apiCache.get(norm);
  1201→}

  1203→function translatePage(langCode: string) {
  1204→  if (isTranslating) return;
  1205→  isTranslating = true;
  1206→  try {
  1207→    currentLang = langCode;
  1208→    const dictionary = dictionaries[langCode];
  1209→    document.documentElement.lang = langCode === "pt" ? "pt-BR" : langCode;
  1210→
  1211→    const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT, {
  1212→      acceptNode(node) {
  1213→        const parent = node.parentElement;
  1214→        if (!parent) return NodeFilter.FILTER_REJECT;
  1215→        if ("SCRIPT STYLE TEXTAREA INPUT".includes(parent.tagName)) return NodeFilter.FILTER_REJECT;
  1216→        if (hasNoTranslateAncestor(node)) return NodeFilter.FILTER_REJECT;
  1217→        return NodeFilter.FILTER_ACCEPT;
  1218→      },
  1219→    });
  1220→
  1221→    const nodes: Text[] = [];
  1222→    let n: Node | null;
  1223→    while ((n = walker.nextNode())) nodes.push(n as Text);
  1224→
  1225→    const untranslated: string[] = [];
  1226→
  1227→    for (const textNode of nodes) {
  1228→      if (hasNoTranslateAncestor(textNode)) continue;
  1229→      if (!originalText.has(textNode)) originalText.set(textNode, textNode.nodeValue || "");
  1230→      const source = originalText.get(textNode) || "";
  1231→      const trimmed = source.trim();
  1232→      if (!trimmed) continue;
  1233→
  1234→      if (langCode === "pt") {
  1235→        if (textNode.nodeValue !== source) textNode.nodeValue = source;
  1236→        continue;
  1237→      }
  1238→
  1239→      const translated = lookup(trimmed, langCode, dictionary);
  1240→      if (translated) {
  1241→        const nextValue = source === trimmed ? translated : source.replace(trimmed, translated);
  1242→        if (textNode.nodeValue !== nextValue) textNode.nodeValue = nextValue;
  1243→      } else if (trimmed.length > 2 && !/^\d+[\,\d]*$/.test(trimmed) && !/^[#\-+°×→←↑↓]*$/.test(trimmed)) {
  1244→        untranslated.push(trimmed);
  1245→      }
  1246→    }
  1247→
  1248→    if (langCode !== "pt") {
  1249→      document.querySelectorAll("input[placeholder], textarea[placeholder]").forEach((el) => {
  1250→        const htmlEl = el as HTMLInputElement;
  1251→        const ph = htmlEl.placeholder;
  1252→        if (!ph) return;
  1253→        const tr = lookup(ph, langCode, dictionary);
  1254→        if (tr) htmlEl.placeholder = tr;
  1255→      });
  1256→      document.querySelectorAll("[title]").forEach((el) => {
  1257→        const htmlEl = el as HTMLElement;
  1258→        const t = htmlEl.getAttribute("title");
  1259→        if (!t) return;
  1260→        const tr = lookup(t, langCode, dictionary);
  1261→        if (tr) htmlEl.setAttribute("title", tr);
  1262→      });
  1263→    }
  1264→
  1265→    if (untranslated.length > 0 && langCode !== "pt") {
  1266→      queueApiTranslation(untranslated, langCode);
  1267→    }
  1268→  } finally {
  1269→    isTranslating = false;
  1270→  }
  1271→}

  1273→function queueApiTranslation(texts: string[], langCode: string) {
  1274→  if (apiDebounceTimer) clearTimeout(apiDebounceTimer);
  1275→  apiDebounceTimer = setTimeout(async () => {
  1276→    const unique = [...new Set(texts)].slice(0, 30);
  1277→    const toFetch = unique.filter((t) => !apiCache.has(t) && !apiCache.has(stripAccents(t)));
  1278→    if (toFetch.length === 0) return;
  1279→    try {
  1280→      const langMap: Record<string, string> = { en: "en", es: "es", fr: "fr", de: "de", ru: "ru" };
  1281→      const targetLang = langMap[langCode] || "en";
  1282→      const res = await fetch("/api/translate", {
  1283→        method: "POST",
  1284→        headers: { "Content-Type": "application/json" },
  1285→        body: JSON.stringify({ texts: toFetch, targetLang }),
  1286→      });
  1287→      if (!res.ok) return;
  1288→      const data = await res.json();
  1289→      const translations: Record<string, string> = data.translations || {};
  1290→      for (const [key, val] of Object.entries(translations)) {
  1291→        if (val && val !== key) {
  1292→          apiCache.set(key, val);
  1293→          const norm = stripAccents(key);
  1294→          if (norm !== key) apiCache.set(norm, val);
  1295→        }
  1296→      }
  1297→      if (Object.keys(translations).length > 0) {
  1298→        translatePage(langCode);
  1299→      }
  1300→    } catch {
  1301→      // Silently fail
  1302→    }
  1303→  }, 400);
  1304→}

  1306→function setLanguage(code: string) {
  1307→  currentLang = code;
  1308→  localStorage.setItem("dayr-language", code);
  1309→  if (code === "pt") {
  1310→    const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT);
  1311→    let n: Node | null;
  1312→    while ((n = walker.nextNode())) {
  1313→      const textNode = n as Text;
  1314→      const orig = originalText.get(textNode);
  1315→      if (orig !== undefined && textNode.nodeValue !== orig) textNode.nodeValue = orig;
  1316→    }
  1317→    document.documentElement.lang = "pt-BR";
  1318→    if (observer) { observer.disconnect(); observer = null; }
  1319→    apiCache.clear();
  1320→    document.querySelectorAll("input[placeholder], textarea[placeholder]").forEach((el) => {
  1321→      const htmlEl = el as HTMLInputElement;
  1322→      const key = htmlEl.getAttribute("data-original-placeholder");
  1323→      if (key) htmlEl.placeholder = key;
  1324→    });
  1325→    return;
  1326→  }
  1327→
  1328→  translatePage(code);
  1329→
  1330→  if (!observer) {
  1331→    let observerDebounce: ReturnType<typeof setTimeout> | null = null;
  1332→    observer = new MutationObserver(() => {
  1333→      if (observerDebounce) clearTimeout(observerDebounce);
  1334→      observerDebounce = setTimeout(() => translatePage(currentLang), 150);
  1335→    });
  1336→    observer.observe(document.body, { childList: true, subtree: true, characterData: true });
  1337→  }
  1338→}
