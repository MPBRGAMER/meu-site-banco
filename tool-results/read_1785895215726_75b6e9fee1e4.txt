    63→function parsePriceValue(price: string): number {
    64→  const parts = price.split(":");
    65→  if (parts.length !== 2) return 9999;
    66→  const qty = parseInt(parts[0]);
    67→  const val = parseInt(parts[1]);
    68→  if (val === 0) return 9999;
    69→  return qty / val;
    70→}

    76→function Sparkline({ itemId }: { itemId: string }) {
    77→  const trend = priceTrends.trends[itemId];
    78→  if (!trend || trend.history.length < 2) return null;
    79→  const data = trend.history;
    80→  const min = Math.min(...data);
    81→  const max = Math.max(...data);
    82→  const range = max - min || 1;
    83→  const width = 50;
    84→  const height = 18;
    85→  const padding = 2;
    86→  const points = data
    87→    .map((value, index) => {
    88→      const x = (index / (data.length - 1)) * (width - padding * 2) + padding;
    89→      const y = height - ((value - min) / range) * (height - padding * 2) - padding;
    90→      return `${x},${y}`;
    91→    })
    92→    .join(" ");
    93→  const color = trend.trend === "up" ? "#22c55e" : trend.trend === "down" ? "#ef4444" : "#64748b";
    94→  return (
    95→    <div className="flex items-center gap-1">
    96→      <svg width={width} height={height} className="shrink-0">
    97→        <polyline points={points} fill="none" stroke={color} strokeWidth="1.5" vectorEffect="non-scaling-stroke" />
    98→      </svg>
    99→      {trend.change !== 0 && (
   100→        <span className={`text-[10px] font-bold flex items-center gap-0.5 ${trend.trend === "up" ? "text-green-400" : "text-red-400"}`}>
   101→          {trend.trend === "up" ? <TrendingUp className="w-2.5 h-2.5" /> : <TrendingDown className="w-2.5 h-2.5" />}
   102→          {trend.change > 0 ? "+" : ""}{trend.change}%
   103→        </span>
   104→      )}
   105→    </div>
   106→  );
   107→}

   408→  const reportMap = useMemo(() => {
   409→    const map: Record<string, { avg: number; count: number; reports: typeof priceReports }> = {};
   410→    for (const r of priceReports) {
   411→      if (!map[r.itemId]) map[r.itemId] = { avg: 0, count: 0, reports: [] };
   412→      map[r.itemId].avg += r.steelPrice;
   413→      map[r.itemId].count++;
   414→      map[r.itemId].reports.push(r);
   415→    }
   416→    for (const key of Object.keys(map)) {
   417→      map[key].avg = Math.round(map[key].avg / map[key].count);
   418→    }
   419→    return map;
   420→  }, [priceReports]);
   421→
   422→  const filteredCategories = useMemo(() => {
   423→    return categories
   424→      .map((cat) => ({
   425→        ...cat,
   426→        items: cat.items
   427→          .filter((item) => {
   428→            if (showOnlyMissing && !isPriceUnknown(item.steel)) return false;
   429→            return item.name.toLowerCase().includes(search.toLowerCase()) || item.id.toLowerCase().includes(search.toLowerCase()) || item.notes.toLowerCase().includes(search.toLowerCase());
   430→          })
   431→          .sort((a, b) => {
   432→            let cmp = 0;
   433→            if (sortField === "name") cmp = a.name.localeCompare(b.name);
   434→            else if (sortField === "steel") cmp = parsePriceValue(a.steel) - parsePriceValue(b.steel);
   435→            else if (sortField === "cement") cmp = parsePriceValue(a.cement) - parsePriceValue(b.cement);
   436→            return sortDir === "asc" ? cmp : -cmp;
   437→          }),
   438→      }))
   439→      .filter((cat) => activeCategory === null || cat.id === activeCategory)
   440→      .filter((cat) => cat.items.length > 0);
   441→  }, [categories, search, activeCategory, sortField, sortDir, showOnlyMissing]);
   442→
   443→  const totalItems = filteredCategories.reduce((sum, c) => sum + c.items.length, 0);
   444→  const totalAllItems = categories.reduce((sum, c) => sum + c.items.length, 0);
   445→  const totalMissing = categories.reduce((sum, c) => sum + c.items.filter((i) => isPriceUnknown(i.steel)).length, 0);
   446→
   447→  const toggleSort = (field: "name" | "steel" | "cement") => {
   448→    if (sortField === field) setSortDir(sortDir === "asc" ? "desc" : "asc");
   449→    else { setSortField(field); setSortDir("asc"); }
   450→  };
   451→
   452→  const toggleCategory = (id: string) => {
   453→    setExpandedCategories((prev) => {
   454→      const next = new Set(prev);
   455→      if (next.has(id)) next.delete(id);
   456→      else next.add(id);
   457→      return next;
   458→    });
   459→  };
   460→
   461→  const expandAll = () => setExpandedCategories(new Set(filteredCategories.map((c) => c.id)));
   462→  const collapseAll = () => setExpandedCategories(new Set());
   463→
   464→  const handleReport = async (d: { itemId: string; itemName: string; nickname: string; steelPrice: number; cementPrice: number }) => {
   465→    await reportPrice(d);
   466→    await fetchReports();
   467→  };
   468→
   469→  return (
   470→    <div className="space-y-3">
   471→      <div className="flex items-center justify-between flex-wrap gap-2">
   472→        <div>
   473→          <h2 className="text-lg font-bold text-primary flex items-center gap-2">Tabela de Precos</h2>
   474→          <p className="text-[10px] text-muted-foreground">{totalAllItems} itens | {totalMissing} sem preco | Fonte: dayr.wiki.gg + comunidade</p>
   475→        </div>
   476→        <div className="flex items-center gap-1.5">
   477→          <button onClick={() => setShowReportar(true)} className="flex items-center gap-1.5 px-2.5 py-1.5 rounded-md text-xs font-medium border border-green-500/30 text-green-400 hover:bg-green-500/10 transition-all">
   478→            <MessageSquarePlus className="w-3.5 h-3.5" /> Reportar
   479→          </button>
   480→          <button onClick={() => setShowEditar(true)} className="flex items-center gap-1.5 px-2.5 py-1.5 rounded-md text-xs font-medium border border-yellow-500/30 text-yellow-400 hover:bg-yellow-500/10 transition-all">
   481→            <Pencil className="w-3.5 h-3.5" /> Editar
   482→          </button>
   483→          <button onClick={() => setShowGuia(true)} className="flex items-center gap-1.5 px-2.5 py-1.5 rounded-md text-xs font-medium border border-blue-500/30 text-blue-400 hover:bg-blue-500/10 transition-all">
   484→            <BookOpen className="w-3.5 h-3.5" /> Guia
   485→          </button>
   486→        </div>
   487→      </div>
   488→
   489→      <div className="rounded-lg border border-border bg-card p-3 grid grid-cols-3 gap-2">
   490→        <div className="flex items-center gap-2">
   491→          <div className="w-7 h-7 rounded-md bg-green-500/10 flex items-center justify-center"><MessageSquarePlus className="w-3.5 h-3.5 text-green-400" /></div>
   492→          <div><p className="text-[10px] text-muted-foreground">Reports</p><p className="text-sm font-bold font-mono text-foreground">{priceReports.length}</p></div>
   493→        </div>
   494→        <div className="flex items-center gap-2">
   495→          <div className="w-7 h-7 rounded-md bg-orange-500/10 flex items-center justify-center"><AlertCircle className="w-3.5 h-3.5 text-orange-400" /></div>
   496→          <div><p className="text-[10px] text-muted-foreground">Sem Preco</p><p className="text-sm font-bold font-mono text-foreground">{totalMissing}</p></div>
   497→        </div>
   498→        <div className="flex items-center gap-2">
   499→          <div className="w-7 h-7 rounded-md bg-blue-500/10 flex items-center justify-center"><BarChart3 className="w-3.5 h-3.5 text-blue-400" /></div>
   500→          <div><p className="text-[10px] text-muted-foreground">Itens com Tendencia</p><p className="text-sm font-bold font-mono text-foreground">{Object.keys(priceTrends.trends).length}</p></div>
   501→        </div>
   502→      </div>
   503→
   504→      <div className="flex items-center gap-2">
   505→        <div className="relative flex-1">
   506→          <Search className="w-4 h-4 text-muted-foreground absolute left-3 top-1/2 -translate-y-1/2" />
   507→          <Input placeholder="Buscar item..." value={search} onChange={(e) => setSearch(e.target.value)} className="pl-9 bg-card border-border text-sm h-9" />
   508→        </div>
   509→        <button onClick={() => setShowOnlyMissing(!showOnlyMissing)} className={`flex items-center gap-1.5 px-2.5 py-1.5 rounded-md text-xs font-medium border transition-all whitespace-nowrap h-9 ${showOnlyMissing ? "bg-orange-500/20 text-orange-400 border-orange-500/30" : "bg-card text-muted-foreground border-border hover:text-foreground"}`}>
   510→          <AlertCircle className="w-3.5 h-3.5" /> Sem Preco
   511→        </button>
   512→        <button onClick={expandAll} className="px-2.5 py-1.5 rounded-md text-xs font-medium border border-border bg-card text-muted-foreground hover:text-foreground transition-all h-9">Abrir</button>
   513→        <button onClick={collapseAll} className="px-2.5 py-1.5 rounded-md text-xs font-medium border border-border bg-card text-muted-foreground hover:text-foreground transition-all h-9">Fechar</button>
   514→      </div>
   515→
   516→      <div className="flex flex-wrap gap-1.5">
   517→        <button onClick={() => setActiveCategory(null)} className={`px-2.5 py-1 rounded-md text-xs font-medium border transition-all ${activeCategory === null ? "bg-primary/20 text-primary border-primary/30" : "bg-card text-muted-foreground border-border hover:text-foreground hover:bg-accent"}`}>Todos ({totalAllItems})</button>
   518→        {categories.map((cat) => (
   519→          <button key={cat.id} onClick={() => setActiveCategory(activeCategory === cat.id ? null : cat.id)} className={`px-2.5 py-1 rounded-md text-xs font-medium border transition-all whitespace-nowrap ${activeCategory === cat.id ? "bg-primary/20 text-primary border-primary/30" : "bg-card text-muted-foreground border-border hover:text-foreground hover:bg-accent"}`}>{cat.name} ({cat.items.length})</button>
   520→        ))}
   521→      </div>
   522→
   523→      <p className="text-[10px] text-muted-foreground">{totalItems} {totalItems === 1 ? "item" : "itens"}{search ? ` para "${search}"` : ""}{showOnlyMissing ? " (sem preco)" : ""}</p>
   524→
   525→      <div className="rounded-lg border border-border bg-card overflow-hidden">
   526→        <div className="grid grid-cols-12 gap-1 px-3 py-2 border-b border-border bg-muted/30 text-[10px] font-bold text-muted-foreground uppercase tracking-wider">
   527→          <div className="col-span-1">Icone</div>
   528→          <button onClick={() => toggleSort("name")} className="col-span-3 flex items-center gap-1 hover:text-foreground transition-colors text-left">Item <ArrowUpDown className="w-3 h-3" /></button>
   529→          <button onClick={() => toggleSort("steel")} className="col-span-2 flex items-center gap-1 hover:text-foreground transition-colors text-left">Aco ($) <ArrowUpDown className="w-3 h-3" /></button>
   530→          <button onClick={() => toggleSort("cement")} className="col-span-2 flex items-center gap-1 hover:text-foreground transition-colors text-left">Cimento <ArrowUpDown className="w-3 h-3" /></button>
   531→          <div className="col-span-1 hidden sm:flex justify-center">Dem.</div>
   532→          <div className="col-span-2 hidden sm:flex justify-center">Tendencia</div>
   533→          <div className="col-span-1 hidden md:flex justify-center">Rep.</div>
   534→        </div>
   535→
   536→        {filteredCategories.length === 0 ? (
   537→          <div className="p-8 text-center text-sm text-muted-foreground">Nenhum item encontrado.</div>
   538→        ) : (
   539→          filteredCategories.map((cat) => (
   540→            <div key={cat.id}>
   541→              <button onClick={() => toggleCategory(cat.id)} className="w-full px-3 py-2.5 border-b border-border bg-primary/5 hover:bg-primary/10 transition-colors text-left">
   542→                <div className="flex items-center gap-2">
   543→                  {expandedCategories.has(cat.id) ? <ChevronDown className="w-3.5 h-3.5 text-primary" /> : <ChevronRight className="w-3.5 h-3.5 text-muted-foreground" />}
   544→                  <span className="text-sm font-bold text-foreground">{cat.name}</span>
   545→                  <span className="text-[10px] text-muted-foreground font-mono">({cat.items.length})</span>
   546→                </div>
   547→              </button>
   548→              {expandedCategories.has(cat.id) && (
   549→                <div>
   550→                  {cat.items.map((item, idx) => {
   551→                    const missing = isPriceUnknown(item.steel);
   552→                    const rep = reportMap[item.id];
   553→                    return (
   554→                      <div key={item.id} className={`grid grid-cols-12 gap-1 px-3 py-2 border-b border-border/50 hover:bg-muted/20 transition-colors text-xs ${idx % 2 === 0 ? "" : "bg-muted/5"} ${missing ? "opacity-70" : ""}`}>
   555→                        <div className="col-span-1 text-center text-base leading-6">{getItemIcon(item.id)}</div>
   556→                        <div className="col-span-3 min-w-0">
   557→                          <p className="font-semibold text-foreground truncate">{item.name}</p>
   558→                          {item.notes && item.notes !== "Preco pendente - reporte para ajudar!" && <p className="text-[10px] text-muted-foreground truncate hidden sm:block">{item.notes}</p>}
   559→                          {missing && <p className="text-[9px] text-orange-400 truncate">Preco pendente</p>}
   560→                        </div>
   561→                        <div className={`col-span-2 font-mono font-bold ${missing ? "text-orange-400/60" : getSteelColor(item.demand)}`}>{item.steel}</div>
   562→                        <div className={`col-span-2 font-mono font-bold ${missing ? "text-orange-400/60" : "text-foreground"}`}>{item.cement}</div>
   563→                        <div className="col-span-1 hidden sm:flex justify-center">
   564→                          <span className={`px-1 py-0.5 rounded text-[9px] font-bold border ${demandColors[item.demand] || demandColors.medium}`}>{demandLabels[item.demand] || "?"}</span>
   565→                        </div>
   566→                        <div className="col-span-2 hidden sm:flex justify-center items-center">
   567→                          {rep ? (
   568→                            <div className="flex items-center gap-1">
   569→                              <span className={`text-[10px] font-bold ${rep.avg > 0 ? "text-primary" : "text-muted-foreground"}`}>
   570→                                {rep.avg > 0 ? rep.avg + "$" : "-"}
   571→                              </span>
   572→                              <span className="text-[9px] text-muted-foreground">({rep.count})</span>
   573→                            </div>
   574→                          ) : (
   575→                            <Sparkline itemId={item.id} />
   576→                          )}
   577→                        </div>
   578→                        <div className="col-span-1 hidden md:flex justify-center items-center">
   579→                          {rep && rep.count > 0 ? (
   580→                            <span className="text-[9px] font-mono text-green-400">{rep.count}</span>
   581→                          ) : (
   582→                            <span className="text-[9px] text-muted-foreground">-</span>
   583→                          )}
   584→                        </div>
   585→                      </div>
   586→                    );
   587→                  })}
   588→                </div>
   589→              )}
   590→            </div>
   591→          ))
   592→        )}
   593→      </div>
   594→
   595→      <div className="rounded-md border border-border bg-card p-3">
   596→        <div className="flex items-center justify-between mb-2">
   597→          <h3 className="text-xs font-bold text-foreground flex items-center gap-2"><Users className="w-3.5 h-3.5 text-primary" /> Ultimos Reports da Comunidade</h3>
   598→        </div>
   599→        {priceReports.length === 0 ? (
   600→          <p className="text-[11px] text-muted-foreground">Nenhum report ainda. Seja o primeiro a reportar precos!</p>
   601→        ) : (
   602→          <div className="space-y-1 max-h-48 overflow-y-auto">
   603→            {priceReports.slice(0, 20).map((r) => (
   604→              <div key={r.id} className="flex items-center justify-between py-1 px-2 rounded border-b border-border/30 text-[11px]">
   605→                <div className="flex items-center gap-2 min-w-0">
   606→                  <span className="text-primary font-semibold shrink-0">{r.nickname}</span>
   607→                  <span className="text-muted-foreground truncate">reportou {r.itemName}</span>
   608→                </div>
   609→                <div className="flex items-center gap-2 shrink-0">
   610→                  <span className="font-mono text-foreground">{r.steelPrice}$</span>
   611→                  <span className="font-mono text-muted-foreground text-[10px]">{new Date(r.data).toLocaleDateString("pt-BR")}</span>
   612→                </div>
   613→              </div>
   614→            ))}
   615→          </div>
   616→        )}
   617→      </div>
   618→
   619→      <ReportarModal isOpen={showReportar} onClose={() => setShowReportar(false)} items={allItems} onReport={handleReport} />
   620→      <EditarModal isOpen={showEditar} onClose={() => setShowEditar(false)} items={allItems} />
   621→      <GuiaModal isOpen={showGuia} onClose={() => setShowGuia(false)} />
   622→    </div>
   623→  );
   624→}
