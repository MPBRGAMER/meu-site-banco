  1144→  1161→function hasNoTranslateAncestor(node: Node): boolean {
  1145→  1162→  let el: Node | null = node.parentElement;
  1146→  1163→  while (el) {
  1147→  1164→    if (el instanceof HTMLElement && el.hasAttribute("data-no-translate")) return true;
  1148→  1165→    el = el.parentElement;
  1149→  1166→  }
  1150→  1167→  return false;
  1151→  1168→}

  1153→  1170→function translatePage(langCode: string) {
  1154→  1171→  currentLang = langCode;
  1155→  1172→  const dictionary = dictionaries[langCode];
  1156→  1173→  document.documentElement.lang = langCode === "pt" ? "pt-BR" : langCode;
  1157→  1174→
  1158→  1175→  const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT, {
  1159→  1176→    acceptNode(node) {
  1160→  1177→      const parent = node.parentElement;
  1161→  1178→      if (!parent) return NodeFilter.FILTER_REJECT;
  1162→  1179→      if ("SCRIPT STYLE TEXTAREA INPUT".includes(parent.tagName)) return NodeFilter.FILTER_REJECT;
  1163→  1180→      if (hasNoTranslateAncestor(node)) return NodeFilter.FILTER_REJECT;
  1164→  1181→      return NodeFilter.FILTER_ACCEPT;
  1165→  1182→    },
  1166→  1183→  });
  1167→  1184→
  1168→  1185→  const nodes: Text[] = [];
  1169→  1186→  let node: Node | null;
  1170→  1187→  while ((node = walker.nextNode())) nodes.push(node as Text);
  1171→  1188→
  1172→  1189→  const untranslated: string[] = [];
  1173→  1190→
  1174→  1191→  for (const textNode of nodes) {
  1175→  1192→    if (!originalText.has(textNode)) originalText.set(textNode, textNode.nodeValue || "");
  1176→  1193→    const source = originalText.get(textNode) || "";
  1177→  1194→    const trimmed = source.trim();
  1178→  1195→    if (!trimmed) continue;
  1179→  1196→
  1180→  1197→    if (langCode === "pt") {
  1181→  1198→      if (textNode.nodeValue !== source) textNode.nodeValue = source;
  1182→  1199→      continue;
  1183→  1200→    }
  1184→  1201→
  1185→  1202→    let translated: string | undefined;
  1186→  1203→    if (langCode === "en") {
  1187→  1204→      translated = EN[trimmed] || apiCache.get(trimmed);
  1188→  1205→    } else {
  1189→  1206→      translated = dictionary?.[trimmed] || apiCache.get(trimmed);
  1190→  1207→    }
  1191→  1208→    if (translated) {
  1192→  1209→      const nextValue = source === trimmed ? translated : source.replace(trimmed, translated);
  1193→  1210→      if (textNode.nodeValue !== nextValue) textNode.nodeValue = nextValue;
  1194→  1211→    } else if (trimmed.length > 2 && !/^\d+[,\d]*$/.test(trimmed) && !/^[#\-+°×→←↑↓]*$/.test(trimmed)) {
  1195→  1212→      untranslated.push(trimmed);
  1196→  1213→    }
  1197→  1214→  }
  1198→  1215→
  1199→  1216→  // Also translate placeholders and titles
  1200→  1217→  if (langCode !== "pt") {
  1201→  1218→    document.querySelectorAll("input[placeholder], textarea[placeholder]").forEach((el) => {
  1202→  1219→      const htmlEl = el as HTMLInputElement;
  1203→  1220→      const ph = htmlEl.placeholder;
  1204→  1221→      if (!ph) return;
  1205→  1222→      const tr = langCode === "en"
  1206→  1223→        ? (EN[ph] || apiCache.get(ph))
  1207→  1224→        : (dictionary?.[ph] || apiCache.get(ph));
  1208→  1225→      if (tr) htmlEl.placeholder = tr;
  1209→  1226→    });
  1210→  1227→    document.querySelectorAll("[title]").forEach((el) => {
  1211→  1228→      const htmlEl = el as HTMLElement;
  1212→  1229→      const t = htmlEl.getAttribute("title");
  1213→  1230→      if (!t) return;
  1214→  1231→      const tr = langCode === "en"
  1215→  1232→        ? (EN[t] || apiCache.get(t))
  1216→  1233→        : (dictionary?.[t] || apiCache.get(t));
  1217→  1234→      if (tr) htmlEl.setAttribute("title", tr);
  1218→  1235→    });
  1219→  1236→  }
  1220→  1237→
  1221→  1238→  // Debounced API fallback
  1222→  1239→  if (untranslated.length > 0 && langCode !== "pt") {
  1223→  1240→    queueApiTranslation(untranslated, langCode);
  1224→  1241→  }
  1225→  1242→}

  1227→  1244→function queueApiTranslation(texts: string[], langCode: string) {
  1228→  1245→  if (apiDebounceTimer) clearTimeout(apiDebounceTimer);
  1229→  1246→  apiDebounceTimer = setTimeout(async () => {
  1230→  1247→    const unique = [...new Set(texts)].slice(0, 30);
  1231→  1248→    const toFetch = unique.filter((t) => !apiCache.has(t));
  1232→  1249→    if (toFetch.length === 0) return;
  1233→  1250→    try {
  1234→  1251→      const langMap: Record<string, string> = { en: "en", es: "es", fr: "fr", de: "de", ru: "ru" };
  1235→  1252→      const targetLang = langMap[langCode] || "en";
  1236→  1253→      const res = await fetch("/api/translate", {
  1237→  1254→        method: "POST",
  1238→  1255→        headers: { "Content-Type": "application/json" },
  1239→  1256→        body: JSON.stringify({ texts: toFetch, targetLang }),
  1240→  1257→      });
  1241→  1258→      if (!res.ok) return;
  1242→  1259→      const data = await res.json();
  1243→  1260→      const translations: Record<string, string> = data.translations || {};
  1244→  1261→      for (const [key, val] of Object.entries(translations)) {
  1245→  1262→        if (val && val !== key) apiCache.set(key, val);
  1246→  1263→      }
  1247→  1264→      // Re-translate with new cache entries
  1248→  1265→      if (Object.keys(translations).length > 0) {
  1249→  1266→        translatePage(langCode);
  1250→  1267→      }
  1251→  1268→    } catch {
  1252→  1269→      // Silently fail - dictionary covers most strings
  1253→  1270→    }
  1254→  1271→  }, 400);
  1255→  1272→}

  1257→  1274→function setLanguage(code: string) {
  1258→  1275→  currentLang = code;
  1259→  1276→  localStorage.setItem("dayr-language", code);
  1260→  1277→  // Reset all text nodes to original before translating

  1262→  1279→    const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT);

  1264→  1281→    while ((node = walker.nextNode())) {
  1265→  1282→      const textNode = node as Text;
  1266→  1283→      const orig = originalText.get(textNode);
  1267→  1284→      if (orig !== undefined && textNode.nodeValue !== orig) textNode.nodeValue = orig;
  1268→  1285→    }
  1269→  1286→    document.documentElement.lang = "pt-BR";
  1270→  1287→    if (observer) { observer.disconnect(); observer = null; }
  1271→  1288→    // Reset placeholders

  1273→  1290→      const htmlEl = el as HTMLInputElement;
  1274→  1291→      const key = htmlEl.getAttribute("data-original-placeholder");
  1275→  1292→      if (key) htmlEl.placeholder = key;
  1276→  1293→    });
  1277→  1294→    return;
  1278→  1295→  }
  1279→  1296→
  1280→  1297→  translatePage(code);
  1281→  1298→
  1282→  1299→  if (!observer) {
  1283→  1300→    observer = new MutationObserver(() => {
  1284→  1301→ translatePage(currentLang);
  1285→  1302→    });
  1286→  1303→    observer.observe(document.body, { childList: true, subtree: true, characterData: true });
  1287→  1304→  }
  1288→  1305→}
