   150→        const itemId = searchParams.get("itemId");
   151→        const where = itemId ? { itemId } : {};
   152→        const data = await db.priceReport.findMany({
   153→          where,
   154→          orderBy: { data: "desc" },
   155→          take: 200,
   156→        });
   157→        return json(data);
   158→      }
   159→      case "getReporterRanking": {
   160→        const reports = await db.priceReport.findMany({
   161→          orderBy: { data: "desc" },
   162→        });
   163→        const grouped: Record<string, { nickname: string; count: number; lastReport: string }> = {};
   164→        for (const r of reports) {
   165→          const key = r.nickname.toLowerCase();
   166→          if (!grouped[key]) {
   167→            grouped[key] = { nickname: r.nickname, count: 0, lastReport: r.data.toISOString() };
   168→          }
   169→          grouped[key].count++;
   170→          if (r.data > new Date(grouped[key].lastReport)) {
   171→            grouped[key].lastReport = r.data.toISOString();
   172→          }
   173→        }
   174→        const ranking = Object.values(grouped)
   175→          .sort((a, b) => b.count - a.count)
   176→          .slice(0, 10);
   177→        return json(ranking);
   178→      }
   179→      case "getAd": {
   180→        const slotId = searchParams.get("slotId");
   181→        if (!slotId) return err("slotId obrigatório");
   182→        const ad = await db.propaganda.findUnique({ where: { slotId } });
   183→        return json({ slotId, codigo: ad?.codigo || "" });
   184→      }
   185→      case "getAllAds": {
   186→        const ads = await db.propaganda.findMany();
   187→        const map: Record<string, string> = {};
   188→        for (const a of ads) {
   189→          if (a.codigo) map[a.slotId] = a.codigo;
   190→        }
   191→        return json(map);
   192→      }
   193→      case "events": {
   194→        // Server-Sent Events for real-time updates
   195→        const stream = new ReadableStream({
   196→          start(controller) {
   197→            const interval = setInterval(() => {
   198→              try {
   199→                controller.enqueue(`data: ${JSON.stringify({ t: Date.now() })}\n\n`);
   200→              } catch {
   201→                clearInterval(interval);
   202→                controller.close();
   203→              }
   204→            }, 5000);
   205→            // Clean up on close
   206→        req.signal.addEventListener("abort", () => {
   207→              clearInterval(interval);
   208→              controller.close();
   209→            });
   210→          },
   211→        });
   212→        return new Response(stream, {
   213→          headers: {
   214→            "Content-Type": "text/event-stream",
   215→            "Cache-Control": "no-cache",
   216→            "Connection": "keep-alive",
   217→          },
   218→        });
   219→      }
   220→      default:
   221→        return err("Ação GET desconhecida: " + action);
   222→    }
   223→  } catch (e: unknown) {
   224→    const msg = e instanceof Error ? e.message : "Erro interno";
   225→    return err(msg, 500);
   226→  }
   227→}

   664→        if (!lotData) return err("Lotérica não encontrada", 404);
   665→        if (numerosVendidos.length === 0) return err("Nenhum número vendido");
   666→
   667→        const numeroSorteado = Math.floor(Math.random() * 1000) + 1;
   668→        const ganhador = numeros.find((n) => n.numero === numeroSorteado && n.comprador);
   669→
   670→        // Calcular premio: acumulado SUBSTITUI o minimo (nao soma)
   671→        // Ex: minimo 100, acumulado 160 → minimo efetivo = 160
   672→        const effectiveMin = Math.max(lotData.premioMinimo, lotData.premioAcumulado || 0);
   673→        const premio80 = lotData.arrecadadoTotal * 0.8;
   674→        const premioFinal = Math.max(premio80, effectiveMin);
   675→        const taxaBanco = lotData.arrecadadoTotal * 0.2;
   676→
   677→        await db.loterica.update({
   678→          where: { id: lotericaId },
   679→          data: {
   680→            status: "sorteio_realizado",
   681→            numeroSorteado,
   682→            ganhador: ganhador ? ganhador.comprador : null,
   683→            valorPremio: premioFinal,
   684→            dataSorteio: new Date(),
   685→          },
   686→        });
   687→
   688→        // Creditar 20% das vendas no estoque do banco
   689→        if (taxaBanco > 0) {
   690→          await db.caixaRegistro.create({
   691→            data: {
   692→              tipo: "entrada",
   693→              descricao: `Taxa bancária Lotérica (20% das vendas)`,
   694→              item: lotData.moedaAceita,
   695→              quantidade: Math.round(taxaBanco),
   696→              valor: Math.round(taxaBanco),
   697→              origem: "loterica",
   698→            },
   699→          });
   700→        }
   701→
   702→        // Se teve ganhador, registrar saida do premio
   703→        if (ganhador) {
   704→          await db.caixaRegistro.create({
   705→            data: {
   706→              tipo: "saida",
   707→              descricao: `Prêmio Lotérica - Número ${numeroSorteado} (${ganhador.comprador})`,
   708→              item: lotData.moedaAceita,
   709→              quantidade: Math.round(premioFinal),
   710→              valor: Math.round(premioFinal),
   711→              origem: "loterica_premio",
   712→            },
   713→          });
   714→        } else {
   715→          // Ninguem acertou - o premio vira o novo minimo para a proxima
   716→          await db.loterica.update({
   717→            where: { id: lotericaId },
   718→            data: { premioAcumulado: Math.round(premioFinal) },
   719→          });
   720→        }
   721→
   722→        return json({
   723→          success: true,
   724→          numeroSorteado,
   725→          ganhador: ganhador ? ganhador.comprador : null,
   726→          premioFinal: Math.round(premioFinal),
   727→          taxaBanco: Math.round(taxaBanco),
   728→          acumulou: !ganhador,
   729→        });
   730→      }
   731→      case "finalizarLoterica": {
   732→        const { lotericaId } = data;
   733→        if (!lotericaId) return err("lotericaId obrigatório");
   734→        const lotData = await db.loterica.findUnique({ where: { id: lotericaId } });
   735→        if (!lotData) return err("Lotérica não encontrada", 404);
   736→        if (lotData.status !== "sorteio_realizado") return err("Sorteio ainda não foi realizado");
   737→
   738→        // Marca como finalizada - pronto para criar proxima
   739→        await db.loterica.update({
   740→          where: { id: lotericaId },
   741→          data: { status: "finalizada" },
   742→        });
   743→
   744→        return json({ success: true, acumuladoProxima: lotData.premioAcumulado || 0 });
   745→      }
   746→
   747→      // === PRICE REPORTS ===
   748→      case "reportPrice": {
   749→        const { itemId, itemName, nickname, steelQty, steelPrice, cementQty, cementPrice } = data;
   750→        if (!itemId || !itemName || !nickname || steelQty === undefined || cementQty === undefined) {
   751→          return err("Campos obrigatorios: itemId, itemName, nickname, steelQty, cementQty");
   752→        }
   753→        const report = await db.priceReport.create({
   754→          data: {
   755→            itemId,
   756→            itemName,
   757→            nickname,
   758→            steelQty: Number(steelQty),
   759→            steelPrice: Number(steelPrice) || 1,
   760→            cementQty: Number(cementQty),
   761→            cementPrice: Number(cementPrice) || 1,
   762→          },
   763→        });
   764→        return json(report);
   765→      }
   766→
   767→      // === PROPAGANDA ===
   768→      case "setAd": {
   769→        const { slotId, codigo } = data;
   770→        if (!slotId) return err("slotId obrigatório");
   771→        const pwd = req.headers.get("x-admin-password");
   772→        if (!pwd || pwd !== ADMIN_PASSWORD) return err("Não autorizado", 403);
   773→        const ad = await db.propaganda.upsert({
   774→          where: { slotId },
   775→          create: { slotId, codigo: codigo || "" },
   776→          update: { codigo: codigo || "" },

   778→        return json({ success: true, id: ad.id });
   779→      }
   780→      case "deleteAd": {
   781→        const { slotId } = data;
   782→        if (!slotId) return err("slotId obrigatório");

   788→
   789→      // === BACKUP ===
   790→      case "backup": {

   793→        const [emprestimos, investidores, tabelasTroca, trocas, comprasVendas, caixa, doadores, leiloes, lances, sorteios, participantes, lotericas, numeros, priceReports, itemOverrides, propagandas, chatSalas, chatMensagens] = await Promise.all([
   794→          db.emprestimo.findMany(),

   827→      }

   829→      default:
   830→        return err("Ação POST desconhecida: " + action);
   831→    }
