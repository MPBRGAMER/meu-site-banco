    67→    67→function parsePriceValue(price: string): number {
    68→    68→  const parts = price.split(":");
    69→    69→  if (parts.length !== 2) return 9999;
    70→    70→  const qty = parseInt(parts[0]);
    71→    71→  const val = parseInt(parts[1]);
    72→    72→  if (val === 0) return 9999;
    73→    73→  return qty / val;
    74→    74→}

    80→    80→function Sparkline({ itemId }: { itemId: string }) {
    81→    81→  const trend = priceTrends.trends[itemId];
    82→    82→  if (!trend || trend.history.length < 2) return null;
    83→    83→  const data = trend.history;
    84→    84→  const min = Math.min(...data);
    85→    85→  const max = Math.max(...data);
    86→    86→  const range = max - min || 1;
    87→    87→  const width = 50;
    88→    88→  const height = 18;
    89→    89→  const padding = 2;
    90→    90→  const points = data
    91→    91→    .map((value, index) => {
    92→    92→      const x = (index / (data.length - 1)) * (width - padding * 2) + padding;
    93→    93→      const y = height - ((value - min) / range) * (height - padding * 2) - padding;
    94→    94→      return `${x},${y}`;
    95→    95→    })
    96→    96→    .join(" ");
    97→    97→  const color = trend.trend === "up" ? "#22c55e" : trend.trend === "down" ? "#ef4444" : "#64748b";
    98→    98→  return (
    99→    99→    <div className="flex items-center gap-1">
   100→   100→      <svg width={width} height={height} className="shrink-0">
   101→   101→        <polyline points={points} fill="none" stroke={color} strokeWidth="1.5" vectorEffect="non-scaling-stroke" />
   102→   102→      </svg>
   103→   103→      {trend.change !== 0 && (
   104→   104→        <span className={`text-[10px] font-bold flex items-center gap-0.5 ${trend.trend === "up" ? "text-green-400" : "text-red-400"}`}>
   105→   105→          {trend.trend === "up" ? <TrendingUp className="w-2.5 h-2.5" /> : <TrendingDown className="w-2.5 h-2.5" />}
   106→   106→          {trend.change > 0 ? "+" : ""}{trend.change}%
   107→   107→        </span>
   108→   108→      )}
   109→   109→    </div>
   110→   110→  );
   111→   111→}
