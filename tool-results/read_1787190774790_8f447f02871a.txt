  1161→  1161→function hasNoTranslateAncestor(node: Node): boolean {
  1162→  1162→  let el: Node | null = node.parentElement;
  1163→  1163→  while (el) {
  1164→  1164→    if (el instanceof HTMLElement && el.hasAttribute("data-no-translate")) return true;
  1165→  1165→    el = el.parentElement;
  1166→  1166→  }
  1167→  1167→  return false;
  1168→  1168→}

  1170→  1170→function translatePage(langCode: string) {
  1171→  1171→  currentLang = langCode;
  1172→  1172→  const dictionary = dictionaries[langCode];
  1173→  1173→  document.documentElement.lang = langCode === "pt" ? "pt-BR" : langCode;
  1174→  1174→
  1175→  1175→  const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT, {
  1176→  1176→    acceptNode(node) {
  1177→  1177→      const parent = node.parentElement;
  1178→  1178→      if (!parent) return NodeFilter.FILTER_REJECT;
  1179→  1179→      if ("SCRIPT STYLE TEXTAREA INPUT".includes(parent.tagName)) return NodeFilter.FILTER_REJECT;
  1180→  1180→      if (hasNoTranslateAncestor(node)) return NodeFilter.FILTER_REJECT;
  1181→  1181→      return NodeFilter.FILTER_ACCEPT;
  1182→  1182→    },
  1183→  1183→  });
  1184→  1184→
  1185→  1185→  const nodes: Text[] = [];
  1186→  1186→  let node: Node | null;
  1187→  1187→  while ((node = walker.nextNode())) nodes.push(node as Text);
  1188→  1188→
  1189→  1189→  const untranslated: string[] = [];
  1190→  1190→
  1191→  1191→  for (const textNode of nodes) {
  1192→  1192→    if (!originalText.has(textNode)) originalText.set(textNode, textNode.nodeValue || "");
  1193→  1193→    const source = originalText.get(textNode) || "";
  1194→  1194→    const trimmed = source.trim();
  1195→  1195→    if (!trimmed) continue;
  1196→  1196→
  1197→  1197→    if (langCode === "pt") {
  1198→  1198→      if (textNode.nodeValue !== source) textNode.nodeValue = source;
  1199→  1199→      continue;
  1200→  1200→    }
  1201→  1201→
  1202→  1202→    let translated: string | undefined;
  1203→  1203→    if (langCode === "en") {
  1204→  1204→      translated = EN[trimmed] || apiCache.get(trimmed);
  1205→  1205→    } else {
  1206→  1206→      translated = dictionary?.[trimmed] || apiCache.get(trimmed);
  1207→  1207→    }
  1208→  1208→    if (translated) {
  1209→  1209→      const nextValue = source === trimmed ? translated : source.replace(trimmed, translated);
  1210→  1210→      if (textNode.nodeValue !== nextValue) textNode.nodeValue = nextValue;
  1211→  1211→    } else if (trimmed.length > 2 && !/^\d+[,\d]*$/.test(trimmed) && !/^[#\-+°×→←↑↓]*$/.test(trimmed)) {
  1212→  1212→      untranslated.push(trimmed);
  1213→  1213→    }
  1214→  1214→  }
  1215→  1215→
  1216→  1216→  // Also translate placeholders and titles
  1217→  1217→  if (langCode !== "pt") {
  1218→  1218→    document.querySelectorAll("input[placeholder], textarea[placeholder]").forEach((el) => {
  1219→  1219→      const htmlEl = el as HTMLInputElement;
  1220→  1220→      const ph = htmlEl.placeholder;
  1221→  1221→      if (!ph) return;
  1222→  1222→      const tr = langCode === "en"
  1223→  1223→        ? (EN[ph] || apiCache.get(ph))
  1224→  1224→        : (dictionary?.[ph] || apiCache.get(ph));
  1225→  1225→      if (tr) htmlEl.placeholder = tr;
  1226→  1226→    });
  1227→  1227→    document.querySelectorAll("[title]").forEach((el) => {
  1228→  1228→      const htmlEl = el as HTMLElement;
  1229→  1229→      const t = htmlEl.getAttribute("title");
  1230→  1230→      if (!t) return;
  1231→  1231→      const tr = langCode === "en"
  1232→  1232→        ? (EN[t] || apiCache.get(t))
  1233→  1233→        : (dictionary?.[t] || apiCache.get(t));
  1234→  1234→      if (tr) htmlEl.setAttribute("title", tr);
  1235→  1235→    });
  1236→  1236→  }
  1237→  1237→
  1238→  1238→  // Debounced API fallback
  1239→  1239→  if (untranslated.length > 0 && langCode !== "pt") {
  1240→  1240→    queueApiTranslation(untranslated, langCode);
  1241→  1241→  }
  1242→  1242→}

  1244→  1244→function queueApiTranslation(texts: string[], langCode: string) {
  1245→  1245→  if (apiDebounceTimer) clearTimeout(apiDebounceTimer);
  1246→  1246→  apiDebounceTimer = setTimeout(async () => {
  1247→  1247→    const unique = [...new Set(texts)].slice(0, 30);
  1248→  1248→    const toFetch = unique.filter((t) => !apiCache.has(t));
  1249→  1249→    if (toFetch.length === 0) return;
  1250→  1250→    try {
  1251→  1251→      const langMap: Record<string, string> = { en: "en", es: "es", fr: "fr", de: "de", ru: "ru" };
  1252→  1252→      const targetLang = langMap[langCode] || "en";
  1253→  1253→      const res = await fetch("/api/translate", {
  1254→  1254→        method: "POST",
  1255→  1255→        headers: { "Content-Type": "application/json" },
  1256→  1256→        body: JSON.stringify({ texts: toFetch, targetLang }),
  1257→  1257→      });
  1258→  1258→      if (!res.ok) return;
  1259→  1259→      const data = await res.json();
  1260→  1260→      const translations: Record<string, string> = data.translations || {};
  1261→  1261→      for (const [key, val] of Object.entries(translations)) {
  1262→  1262→        if (val && val !== key) apiCache.set(key, val);
  1263→  1263→      }
  1264→  1264→      // Re-translate with new cache entries
  1265→  1265→      if (Object.keys(translations).length > 0) {
  1266→  1266→        translatePage(langCode);
  1267→  1267→      }
  1268→  1268→    } catch {
  1269→  1269→      // Silently fail - dictionary covers most strings
  1270→  1270→    }
  1271→  1271→  }, 400);
  1272→  1272→}

  1274→  1274→function setLanguage(code: string) {
  1275→  1275→  currentLang = code;
  1276→  1276→  localStorage.setItem("dayr-language", code);
  1277→  1277→  // Reset all text nodes to original before translating
  1278→  1278→  if (code === "pt") {
  1279→  1279→    const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT);
  1280→  1280→    let node: Node | null;
  1281→  1281→    while ((node = walker.nextNode())) {
  1282→  1282→      const textNode = node as Text;
  1283→  1283→      const orig = originalText.get(textNode);
  1284→  1284→      if (orig !== undefined && textNode.nodeValue !== orig) textNode.nodeValue = orig;
  1285→  1285→    }
  1286→  1286→    document.documentElement.lang = "pt-BR";
  1287→  1287→    if (observer) { observer.disconnect(); observer = null; }
  1288→  1288→    // Reset placeholders
  1289→  1289→    document.querySelectorAll("input[placeholder], textarea[placeholder]").forEach((el) => {
  1290→  1290→      const htmlEl = el as HTMLInputElement;
  1291→  1291→      const key = htmlEl.getAttribute("data-original-placeholder");
  1292→  1292→      if (key) htmlEl.placeholder = key;
  1293→  1293→    });
  1294→  1294→    return;
  1295→  1295→  }
  1296→  1296→
  1297→  1297→  translatePage(code);
  1298→  1298→
  1299→  1299→  if (!observer) {
  1300→  1300→    observer = new MutationObserver(() => {
  1301→  1301→ translatePage(currentLang);
  1302→  1302→    });
  1303→  1303→    observer.observe(document.body, { childList: true, subtree: true, characterData: true });
  1304→  1304→  }
  1305→  1305→}
