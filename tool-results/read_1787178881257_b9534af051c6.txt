   789→   789→const originalText = new WeakMap<Text, string>();
   790→   790→let observer: MutationObserver | null = null;

   793→   793→const apiCache = new Map<string, string>();
   794→   794→
   795→   795→function hasNoTranslateAncestor(node: Node): boolean {
   796→   796→  let el: Node | null = node.parentElement;
   797→   797→  while (el) {
   798→   798→    if (el instanceof HTMLElement && el.hasAttribute("data-no-translate")) return true;
   799→   799→    el = el.parentElement;
   800→   800→  }
   801→   801→  return false;
   802→   802→}

   804→   804→function translatePage(langCode: string) {
   805→   805→  currentLang = langCode;
   806→   806→  const dictionary = dictionaries[langCode];
   807→   807→  document.documentElement.lang = langCode === "pt" ? "pt-BR" : langCode;
   808→   808→
   809→   809→  const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT, {
   810→   810→    acceptNode(node) {
   811→   811→      const parent = node.parentElement;
   812→   812→      if (!parent) return NodeFilter.FILTER_REJECT;
   813→   813→      if ("SCRIPT STYLE TEXTAREA INPUT".includes(parent.tagName)) return NodeFilter.FILTER_REJECT;
   814→   814→      if (hasNoTranslateAncestor(node)) return NodeFilter.FILTER_REJECT;
   815→   815→      return NodeFilter.FILTER_ACCEPT;
   816→   816→    },
   817→   817→  });
   818→   818→
   819→   819→  const nodes: Text[] = [];
   820→   820→  let node: Node | null;
   821→   821→  while ((node = walker.nextNode())) nodes.push(node as Text);
   822→   822→
   823→   823→  const untranslated: string[] = [];
   824→   824→
   825→   825→  for (const textNode of nodes) {
   826→   826→    if (!originalText.has(textNode)) originalText.set(textNode, textNode.nodeValue || "");
   827→   827→    const source = originalText.get(textNode) || "";
   828→   828→    const trimmed = source.trim();
   829→   829→    if (!trimmed) continue;
   830→   830→
   831→   831→    if (langCode === "pt") {
   832→   832→      if (textNode.nodeValue !== source) textNode.nodeValue = source;
   833→   833→      continue;
   834→   834→    }
   835→   835→
   836→   836→    const translated = dictionary?.[trimmed] || apiCache.get(trimmed);
   837→   837→    if (translated) {
   838→   838→      const nextValue = source === trimmed ? translated : source.replace(trimmed, translated);
   839→   839→      if (textNode.nodeValue !== nextValue) textNode.nodeValue = nextValue;
   840→   840→    } else if (trimmed.length > 2 && !/^\d+[,\d]*$/.test(trimmed) && !/^[#\-+°×→←↑↓]*$/.test(trimmed)) {
   841→   841→      untranslated.push(trimmed);
   842→   842→    }
   843→   843→  }
   844→   844→
   845→   845→  // Also translate placeholders and titles
   846→   846→  if (langCode !== "pt") {
   847→   847→    document.querySelectorAll("input[placeholder], textarea[placeholder]").forEach((el) => {
   848→   848→      const htmlEl = el as HTMLInputElement;
   849→   849→      const ph = htmlEl.placeholder;
   850→   850→      if (!ph) return;
   851→   851→      const tr = dictionary?.[ph] || apiCache.get(ph);
   852→   852→      if (tr) htmlEl.placeholder = tr;
   853→   853→    });
   854→   854→    document.querySelectorAll("[title]").forEach((el) => {
   855→   855→      const htmlEl = el as HTMLElement;
   856→   856→      const t = htmlEl.getAttribute("title");
   857→   857→      if (!t) return;
   858→   858→      const tr = dictionary?.[t] || apiCache.get(t);
   859→   859→      if (tr) htmlEl.setAttribute("title", tr);
   860→   860→    });
   861→   861→  }
   862→   862→
   863→   863→  // Debounced API fallback
   864→   864→  if (untranslated.length > 0 && langCode !== "pt") {
   865→   865→    queueApiTranslation(untranslated, langCode);
   866→   866→  }
   867→   867→}
   868→   868→
   869→   869→function queueApiTranslation(texts: string[], langCode: string) {
   870→   870→  if (apiDebounceTimer) clearTimeout(apiDebounceTimer);
   871→   871→  apiDebounceTimer = setTimeout(async () => {
   872→   872→    const unique = [...new Set(texts)].slice(0, 30);
   873→   873→    const toFetch = unique.filter((t) => !apiCache.has(t));
   874→   874→    if (toFetch.length === 0) return;
   875→   875→    try {
   876→   876→      const langMap: Record<string, string> = { en: "en", es: "es", fr: "fr", de: "de", ru: "ru" };
   877→   877→      const targetLang = langMap[langCode] || "en";
   878→   878→      const res = await fetch("/api/translate", {
   879→   879→        method: "POST",
   880→   880→        headers: { "Content-Type": "application/json" },
   881→   881→        body: JSON.stringify({ texts: toFetch, targetLang }),
   882→   882→      });
   883→   883→      if (!res.ok) return;
   884→   884→      const data = await res.json();
   885→   885→      const translations: Record<string, string> = data.translations || {};
   886→   886→      for (const [key, val] of Object.entries(translations)) {
   887→   887→        if (val && val !== key) apiCache.set(key, val);
   888→   888→      }
   889→   889→      // Re-translate with new cache entries
   890→   890→      if (Object.keys(translations).length > 0) {
   891→   891→        translatePage(langCode);
   892→   892→      }
   893→   893→    } catch {
   894→   894→      // Silently fail - dictionary covers most strings
   895→   895→    }
   896→   896→  }, 400);
   897→   897→}

   899→   899→function setLanguage(code: string) {
   900→   900→  currentLang = code;
   901→   901→  localStorage.setItem("dayr-language", code);
   902→   902→  // Reset all text nodes to original before translating
   903→   903→  if (code === "pt") {
   904→   904→    const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT);
   905→   905→    let node: Node | null;
   906→   906→    while ((node = walker.nextNode())) {
   907→   907→      const textNode = node as Text;
   908→   908→      const orig = originalText.get(textNode);
   909→   909→      if (orig !== undefined && textNode.nodeValue !== orig) textNode.nodeValue = orig;
   910→   910→    }
   911→   911→    document.documentElement.lang = "pt-BR";
   912→   912→    if (observer) { observer.disconnect(); observer = null; }
   913→   913→    // Reset placeholders
   914→   914→    document.querySelectorAll("input[placeholder], textarea[placeholder]").forEach((el) => {
   915→   915→      const htmlEl = el as HTMLInputElement;
   916→   916→      const key = htmlEl.getAttribute("data-original-placeholder");
   917→   917→      if (key) htmlEl.placeholder = key;
   918→   918→    });
   919→   919→    return;
   920→   920→  }
   921→   921→
   922→   922→  translatePage(code);
   923→   923→
   924→   924→  if (!observer) {
   925→   925→    observer = new MutationObserver(() => {
   926→   926→ translatePage(currentLang);
   927→   927→    });
   928→   928→    observer.observe(document.body, { childList: true, subtree: true, characterData: true });
   929→   929→  }
   930→   930→}
