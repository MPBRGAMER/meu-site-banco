  1085→   TRANSLATION ENGINE

  1087→const originalText = new WeakMap<Text, string>();
  1088→let observer: MutationObserver | null = null;
  1089→let currentLang = "pt";
  1090→let apiDebounceTimer: ReturnType<typeof setTimeout> | null = null;
  1091→const apiCache = new Map<string, string>();
  1092→
  1093→function hasNoTranslateAncestor(node: Node): boolean {
  1094→  let el: Node | null = node.parentElement;
  1095→  while (el) {
  1096→    if (el instanceof HTMLElement && el.hasAttribute("data-no-translate")) return true;
  1097→    el = el.parentElement;
  1098→  }
  1099→  return false;
  1100→}
  1101→
  1102→function translatePage(langCode: string) {
  1103→  currentLang = langCode;
  1104→  const dictionary = dictionaries[langCode];
  1105→  document.documentElement.lang = langCode === "pt" ? "pt-BR" : langCode;
  1106→
  1107→  const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT, {
  1108→    acceptNode(node) {
  1109→      const parent = node.parentElement;
  1110→      if (!parent) return NodeFilter.FILTER_REJECT;
  1111→      if ("SCRIPT STYLE TEXTAREA INPUT".includes(parent.tagName)) return NodeFilter.FILTER_REJECT;
  1112→      if (hasNoTranslateAncestor(node)) return NodeFilter.FILTER_REJECT;
  1113→      return NodeFilter.FILTER_ACCEPT;
  1114→    },
  1115→  });
  1116→
  1117→  const nodes: Text[] = [];
  1118→  let node: Node | null;
  1119→  while ((node = walker.nextNode())) nodes.push(node as Text);
  1120→
  1121→  const untranslated: string[] = [];
  1122→
  1123→  for (const textNode of nodes) {
  1124→    if (!originalText.has(textNode)) originalText.set(textNode, textNode.nodeValue || "");
  1125→    const source = originalText.get(textNode) || "";
  1126→    const trimmed = source.trim();
  1127→    if (!trimmed) continue;
  1128→
  1129→    if (langCode === "pt") {
  1130→      if (textNode.nodeValue !== source) textNode.nodeValue = source;
  1131→      continue;
  1132→    }
  1133→
  1134→    const translated = dictionary?.[trimmed] || (langCode !== "en" ? EN[trimmed] : null) || apiCache.get(trimmed);
  1135→    if (translated) {
  1136→      const nextValue = source === trimmed ? translated : source.replace(trimmed, translated);
  1137→      if (textNode.nodeValue !== nextValue) textNode.nodeValue = nextValue;
  1138→    } else if (trimmed.length > 2 && !/^\d+[,\d]*$/.test(trimmed) && !/^[#\-+°×→←↑↓]*$/.test(trimmed)) {
  1139→      untranslated.push(trimmed);
  1140→    }
  1141→  }
  1142→
  1143→  // Also translate placeholders and titles
  1144→  if (langCode !== "pt") {
  1145→    document.querySelectorAll("input[placeholder], textarea[placeholder]").forEach((el) => {
  1146→      const htmlEl = el as HTMLInputElement;
  1147→      const ph = htmlEl.placeholder;
  1148→      if (!ph) return;
  1149→      const tr = dictionary?.[ph] || (langCode !== "en" ? EN[ph] : null) || apiCache.get(ph);
  1150→      if (tr) htmlEl.placeholder = tr;
  1151→    });
  1152→    document.querySelectorAll("[title]").forEach((el) => {
  1153→      const htmlEl = el as HTMLElement;
  1154→      const t = htmlEl.getAttribute("title");
  1155→      if (!t) return;
  1156→      const tr = dictionary?.[t] || (langCode !== "en" ? EN[t] : null) || apiCache.get(t);
  1157→      if (tr) htmlEl.setAttribute("title", tr);
  1158→    });
  1159→  }
  1160→
  1161→  // Debounced API fallback
  1162→  if (untranslated.length > 0 && langCode !== "pt") {
  1163→    queueApiTranslation(untranslated, langCode);
  1164→  }
  1165→}
  1166→
  1167→function queueApiTranslation(texts: string[], langCode: string) {
  1168→  if (apiDebounceTimer) clearTimeout(apiDebounceTimer);
  1169→  apiDebounceTimer = setTimeout(async () => {
  1170→    const unique = [...new Set(texts)].slice(0, 30);
  1171→    const toFetch = unique.filter((t) => !apiCache.has(t));
  1172→    if (toFetch.length === 0) return;
  1173→    try {
  1174→      const langMap: Record<string, string> = { en: "en", es: "es", fr: "fr", de: "de", ru: "ru" };
  1175→      const targetLang = langMap[langCode] || "en";
  1176→      const res = await fetch("/api/translate", {
  1177→        method: "POST",
  1178→        headers: { "Content-Type": "application/json" },
  1179→        body: JSON.stringify({ texts: toFetch, targetLang }),
  1180→      });
  1181→      if (!res.ok) return;
  1182→      const data = await res.json();
  1183→      const translations: Record<string, string> = data.translations || {};
  1184→      for (const [key, val] of Object.entries(translations)) {
  1185→        if (val && val !== key) apiCache.set(key, val);
  1186→      }
  1187→      // Re-translate with new cache entries
  1188→      if (Object.keys(translations).length > 0) {
  1189→        translatePage(langCode);
  1190→      }
  1191→    } catch {
  1192→      // Silently fail - dictionary covers most strings
  1193→    }
  1194→  }, 400);
  1195→}

  1197→function setLanguage(code: string) {
  1198→  currentLang = code;
  1199→  localStorage.setItem("dayr-language", code);
  1200→  // Reset all text nodes to original before translating
  1201→  if (code === "pt") {
  1202→    const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT);
  1203→    let node: Node | null;
  1204→    while ((node = walker.nextNode())) {
  1205→      const textNode = node as Text;
  1206→      const orig = originalText.get(textNode);
  1207→      if (orig !== undefined && textNode.nodeValue !== orig) textNode.nodeValue = orig;
  1208→    }
  1209→    document.documentElement.lang = "pt-BR";
  1210→    if (observer) { observer.disconnect(); observer = null; }
  1211→    // Reset placeholders
  1212→    document.querySelectorAll("input[placeholder], textarea[placeholder]").forEach((el) => {
  1213→      const htmlEl = el as HTMLInputElement;
  1214→      const key = htmlEl.getAttribute("data-original-placeholder");
  1215→      if (key) htmlEl.placeholder = key;
  1216→    });
  1217→    return;
  1218→  }
  1219→
  1220→  translatePage(code);
  1221→
  1222→  if (!observer) {
  1223→    observer = new MutationObserver(() => {
  1224→ translatePage(currentLang);
  1225→    });
  1226→    observer.observe(document.body, { childList: true, subtree: true, characterData: true });
  1227→  }
  1228→}
