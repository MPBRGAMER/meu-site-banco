   696→const dictionaries: Record<string, Record<string, string>> = { en: EN, es: ES, fr: FR, de: DE, ru: RU };
   697→const originalText = new WeakMap<Text, string>();
   698→let observer: MutationObserver | null = null;
   699→
   700→function translatePage(langCode: string) {
   701→  const dictionary = dictionaries[langCode];
   702→  document.documentElement.lang = langCode === "pt" ? "pt-BR" : langCode;
   703→  const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT, {
   704→    acceptNode(node) {
   705→      const parent = node.parentElement;
   706→      if (!parent || ["SCRIPT", "STYLE", "TEXTAREA", "INPUT"].includes(parent.tagName)) return NodeFilter.FILTER_REJECT;
   707→      return NodeFilter.FILTER_ACCEPT;
   708→    },
   709→  });
   710→  const nodes: Text[] = [];
   711→  let node: Node | null;
   712→  while ((node = walker.nextNode())) nodes.push(node as Text);
   713→  for (const textNode of nodes) {
   714→    if (!originalText.has(textNode)) originalText.set(textNode, textNode.nodeValue || "");
   715→    const source = originalText.get(textNode) || "";
   716→    const trimmed = source.trim();
   717→    if (!trimmed) continue;
   718→    const translated = langCode === "pt" ? source : dictionary?.[trimmed] || source;
   719→    const nextValue = source === trimmed ? translated : source.replace(trimmed, translated);
   720→    if (textNode.nodeValue !== nextValue) textNode.nodeValue = nextValue;
   721→  }
   722→}
   723→
   724→function setLanguage(code: string) {
   725→  localStorage.setItem("dayr-language", code);
   726→  translatePage(code);
   727→  if (!observer) {
   728→    observer = new MutationObserver(() => translatePage(localStorage.getItem("dayr-language") || "pt"));
   729→    observer.observe(document.body, { childList: true, subtree: true });
   730→  }
   731→}
