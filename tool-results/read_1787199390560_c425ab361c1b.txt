   541→  1390→function lookup(text: string, langCode: string, dictionary: Record<string, string> | undefined): string | undefined {

   549→  1398→function translatePage(langCode: string) {
   550→  1399→  if (isTranslating) return;
   551→  1400→  isTranslating = true;
   552→  1401→  try {
   553→  1402→    currentLang = langCode;
   554→  1403→    const dictionary = dictionaries[langCode];
   555→  1404→    document.documentElement.lang = langCode === "pt" ? "pt-BR" : langCode;
   556→  1405→
   557→  1406→    const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT, {
   558→  1407→      acceptNode(node) {
   559→  1408→        const parent = node.parentElement;
   560→  1409→        if (!parent) return NodeFilter.FILTER_REJECT;
   561→  1410→        if ("SCRIPT STYLE TEXTAREA INPUT".includes(parent.tagName)) return NodeFilter.FILTER_REJECT;
   562→  1411→        if (hasNoTranslateAncestor(node)) return NodeFilter.FILTER_REJECT;
   563→  1412→        return NodeFilter.FILTER_ACCEPT;
   564→  1413→      },
   565→  1414→    });
   566→  1415→
   567→  1416→    const nodes: Text[] = [];
   568→  1417→    let n: Node | null;
   569→  1418→    while ((n = walker.nextNode())) nodes.push(n as Text);
   570→  1419→
   571→  1420→    const untranslated: string[] = [];
   572→  1421→
   573→  1422→    for (const textNode of nodes) {
   574→  1423→      if (hasNoTranslateAncestor(textNode)) continue;
   575→  1424→      if (!originalText.has(textNode)) originalText.set(textNode, textNode.nodeValue || "");
   576→  1425→      const source = originalText.get(textNode) || "";
   577→  1426→      const trimmed = source.trim();
   578→  1427→      if (!trimmed) continue;
   579→  1428→
   580→  1429→      if (langCode === "pt") {
   581→  1430→        if (textNode.nodeValue !== source) textNode.nodeValue = source;
   582→  1431→        continue;
   583→  1432→      }
   584→  1433→
   585→  1434→      const translated = lookup(trimmed, langCode, dictionary);
   586→  1435→      if (translated) {
   587→  1436→        const nextValue = source === trimmed ? translated : source.replace(trimmed, translated);
   588→  1437→        if (textNode.nodeValue !== nextValue) textNode.nodeValue = nextValue;
   589→  1438→      } else if (trimmed.length > 2 && !/^\d+[\,\d]*$/.test(trimmed) && !/^[#\-+°×→←↑↓]*$/.test(trimmed)) {
   590→  1439→        untranslated.push(trimmed);
   591→  1440→      }
   592→  1441→    }
   593→  1442→
   594→  1443→    if (langCode !== "pt") {
   595→  1444→      document.querySelectorAll("input[placeholder], textarea[placeholder]").forEach((el) => {
   596→  1445→        const htmlEl = el as HTMLInputElement;
   597→  1446→        const ph = htmlEl.placeholder;
   598→  1447→        if (!ph) return;
   599→  1448→        const tr = lookup(ph, langCode, dictionary);
   600→  1449→        if (tr) htmlEl.placeholder = tr;
   601→  1450→      });
   602→  1451→      document.querySelectorAll("[title]").forEach((el) => {
   603→  1452→        const htmlEl = el as HTMLElement;
   604→  1453→        const t = htmlEl.getAttribute("title");
   605→  1454→        if (!t) return;
   606→  1455→        const tr = lookup(t, langCode, dictionary);
   607→  1456→        if (tr) htmlEl.setAttribute("title", tr);
   608→  1457→      });
   609→  1458→    }
   610→  1459→
   611→  1460→    if (untranslated.length > 0 && langCode !== "pt") {
   612→  1461→      queueApiTranslation(untranslated, langCode);
   613→  1462→    }
   614→  1463→  } finally {
   615→  1464→    isTranslating = false;
   616→  1465→  }
   617→  1466→}

   619→  1468→function queueApiTranslation(texts: string[], langCode: string) {
   620→  1469→  if (apiDebounceTimer) clearTimeout(apiDebounceTimer);
   621→  1470→  apiDebounceTimer = setTimeout(async () => {
   622→  1471→    const unique = [...new Set(texts)].slice(0, 30);
   623→  1472→    const toFetch = unique.filter((t) => !apiCache.has(t) && !apiCache.has(stripAccents(t)));
   624→  1473→    if (toFetch.length === 0) return;
   625→  1474→    try {
   626→  1475→      const langMap: Record<string, string> = { en: "en", es: "es", fr: "fr", de: "de", ru: "ru" };
   627→  1476→      const targetLang = langMap[langCode] || "en";
   628→  1477→      const res = await fetch("/api/translate", {
   629→  1478→        method: "POST",
   630→  1479→        headers: { "Content-Type": "application/json" },
   631→  1480→        body: JSON.stringify({ texts: toFetch, targetLang }),
   632→  1481→      });
   633→  1482→      if (!res.ok) return;
   634→  1483→      const data = await res.json();
   635→  1484→      const translations: Record<string, string> = data.translations || {};
   636→  1485→      for (const [key, val] of Object.entries(translations)) {
   637→  1486→        if (val && val !== key) {
   638→  1487→          apiCache.set(key, val);
   639→  1488→          const norm = stripAccents(key);
   640→  1489→          if (norm !== key) apiCache.set(norm, val);
   641→  1490→        }
   642→  1491→      }
   643→  1492→      if (Object.keys(translations).length > 0) {
   644→  1493→        translatePage(langCode);
   645→  1494→      }
   646→  1495→    } catch {
   647→  1496→      // Silently fail
   648→  1497→    }
   649→  1498→  }, 400);
   650→  1499→}
