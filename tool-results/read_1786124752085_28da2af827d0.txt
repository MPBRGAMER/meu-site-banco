    68→function parsePriceValue(price: string): number {
    69→  const parts = price.split(":");
    70→  if (parts.length !== 2) return 9999;
    71→  const qty = parseInt(parts[0]);
    72→  const val = parseInt(parts[1]);
    73→  if (val === 0) return 9999;
    74→  return qty / val;
    75→}

    81→function Sparkline({ itemId }: { itemId: string }) {
    82→  const trend = priceTrends.trends[itemId];
    83→  if (!trend || trend.history.length < 2) return null;
    84→  const data = trend.history;
    85→  const min = Math.min(...data);
    86→  const max = Math.max(...data);
    87→  const range = max - min || 1;
    88→  const width = 50;
    89→  const height = 18;
    90→  const padding = 2;
    91→  const points = data
    92→    .map((value, index) => {
    93→      const x = (index / (data.length - 1)) * (width - padding * 2) + padding;
    94→      const y = height - ((value - min) / range) * (height - padding * 2) - padding;
    95→      return `${x},${y}`;
    96→    })
    97→    .join(" ");
    98→  const color = trend.trend === "up" ? "#22c55e" : trend.trend === "down" ? "#ef4444" : "#64748b";
    99→  return (
   100→    <div className="flex items-center gap-1">
   101→      <svg width={width} height={height} className="shrink-0">
   102→        <polyline points={points} fill="none" stroke={color} strokeWidth="1.5" vectorEffect="non-scaling-stroke" />
   103→      </svg>
   104→      {trend.change !== 0 && (
   105→        <span className={`text-[10px] font-bold flex items-center gap-0.5 ${trend.trend === "up" ? "text-green-400" : "text-red-400"}`}>
   106→          {trend.trend === "up" ? <TrendingUp className="w-2.5 h-2.5" /> : <TrendingDown className="w-2.5 h-2.5" />}
   107→          {trend.change > 0 ? "+" : ""}{trend.change}%
   108→        </span>
   109→      )}
   110→    </div>
   111→  );
   112→}
