     9→     9→  return NextResponse.json({ error: msg }, { status });

    14→    14→export async function GET(req: NextRequest) {
    15→    15→  const { searchParams } = req.nextUrl;
    16→    16→  const action = searchParams.get("action");
    17→    17→
    18→    18→  try {
    19→    19→    switch (action) {
    20→    20→      case "verifyAdmin": {
    21→    21→        const pwd = searchParams.get("password");
    22→    22→        if (!pwd) return err("Senha obrigatória", 401);
    23→    23→        if (pwd !== ADMIN_PASSWORD) return err("Senha incorreta", 403);
    24→    24→        return json({ success: true });
    25→    25→      }
    26→    26→      case "listEmprestimos": {
    27→    27→        const data = await db.emprestimo.findMany({
    28→    28→          orderBy: { dataEmprestimo: "desc" },
    29→    29→        });
    30→    30→        return json(data);
    31→    31→      }
    32→    32→      case "listInvestidores": {
    33→    33→        const data = await db.investidor.findMany({
    34→    34→          where: { status: "ativo" },
    35→    35→          orderBy: { ordem: "desc" },
    36→    36→        });
    37→    37→        return json(data);
    38→    38→      }
    39→    39→      case "listTabelasTroca": {
    40→    40→        const data = await db.tabelaTroca.findMany();
    41→    41→        return json(data);
    42→    42→      }
    43→    43→      case "listTrocas": {
    44→    44→        const data = await db.trocaRegistro.findMany({
    45→    45→          orderBy: { data: "desc" },
    46→    46→        });
    47→    47→        return json(data);
    48→    48→      }
    49→    49→      case "listComprasVendas": {
    50→    50→        const data = await db.compraVenda.findMany({
    51→    51→          orderBy: { data: "desc" },
    52→    52→        });
    53→    53→        return json(data);
    54→    54→      }
    55→    55→      case "listCaixa": {
    56→    56→        const data = await db.caixaRegistro.findMany({
    57→    57→          orderBy: { data: "desc" },
    58→    58→        });
    59→    59→        return json(data);
    60→    60→      }
    61→    61→      case "listDoadores": {
    62→    62→        const data = await db.doador.findMany({
    63→    63→          orderBy: { ordem: "desc" },
    64→    64→        });
    65→    65→        return json(data);
    66→    66→      }
    67→    67→      case "listLeiloes": {
    68→    68→        const data = await db.leilao.findMany({
    69→    69→          orderBy: { dataCriacao: "desc" },
    70→    70→        });
    71→    71→        return json(data);
    72→    72→      }
    73→    73→      case "listLances": {
    74→    74→        const leilaoId = searchParams.get("leilaoId");
    75→    75→        if (!leilaoId) return err("leilaoId obrigatório");
    76→    76→        const data = await db.lance.findMany({
    77→    77→          where: { leilaoId },
    78→    78→          orderBy: { valor: "desc" },
    79→    79→        });
    80→    80→        return json(data);
    81→    81→      }
    82→    82→      case "listAllLances": {
    83→    83→        const data = await db.lance.findMany({
    84→    84→          orderBy: { valor: "desc" },
    85→    85→        });
    86→    86→        return json(data);
    87→    87→      }
    88→    88→      case "listSorteios": {

    90→    90→          orderBy: { dataCriacao: "desc" },
    91→    91→        });
    92→    92→        return json(data);
    93→    93→      }
    94→    94→      case "getHistoricoSorteios": {
    95→    95→        const sorteios = await db.sorteio.findMany({
    96→    96→          where: { status: "finalizado" },
    97→    97→          orderBy: { dataCriacao: "desc" },
    98→    98→        });
    99→    99→        const historico = await Promise.all(
   100→   100→          sorteios.map(async (s) => {
   101→   101→            const count = await db.participanteSorteio.count({
   102→   102→              where: { sorteioId: s.id },
   103→   103→            });
   104→   104→            return { ...s, totalParticipantes: count };
   105→   105→          })
   106→   106→        );
   107→   107→        return json(historico);
   108→   108→      }
   109→   109→      case "listParticipantes": {
   110→   110→        const sorteioId = searchParams.get("sorteioId");
   111→   111→        if (!sorteioId) return err("sorteioId obrigatório");
   112→   112→        const data = await db.participanteSorteio.findMany({
   113→   113→          where: { sorteioId },
   114→   114→          orderBy: { data: "desc" },
   115→   115→        });
   116→   116→        return json(data);
   117→   117→      }
   118→   118→      case "getLoterica": {
   119→   119→        const data = await db.loterica.findMany({
   120→   120→          where: { status: { in: ["vendas_abertas", "sorteio_realizado"] } },
   121→   121→          orderBy: { dataCriacao: "desc" },
   122→   122→          take: 1,
   123→   123→        });
   124→   124→        return json(data);
   125→   125→      }
   126→   126→      case "getAllLoterica": {
   127→   127→        const data = await db.loterica.findMany({
   128→   128→          orderBy: { dataCriacao: "desc" },
   129→   129→        });
   130→   130→        const historico = await Promise.all(
   131→   131→          data.map(async (l) => {
   132→   132→            const numsVendidos = await db.numeroLoterica.count({
   133→   133→              where: { lotericaId: l.id, comprador: { not: null } },
   134→   134→            });
   135→   135→            return { ...l, totalVendidos: numsVendidos };
   136→   136→          })
   137→   137→        );
   138→   138→        return json(historico);
   139→   139→      }
   140→   140→      case "getNumerosLoterica": {
   141→   141→        const lotericaId = searchParams.get("lotericaId");
   142→   142→        if (!lotericaId) return err("lotericaId obrigatório");
   143→   143→        const data = await db.numeroLoterica.findMany({
   144→   144→          where: { lotericaId },
   145→   145→          orderBy: { numero: "asc" },
   146→   146→        });
   147→   147→        return json(data);
   148→   148→      }
   149→   149→      case "getPriceReports": {
   150→   150→        const itemId = searchParams.get("itemId");
   151→   151→        const where = itemId ? { itemId } : {};
   152→   152→        const data = await db.priceReport.findMany({
   153→   153→          where,
   154→   154→          orderBy: { data: "desc" },
   155→   155→          take: 200,
   156→   156→        });
   157→   157→        return json(data);
   158→   158→      }
   159→   159→      case "getReporterRanking": {
   160→   160→        const reports = await db.priceReport.findMany({
   161→   161→          orderBy: { data: "desc" },
   162→   162→        });
   163→   163→        const grouped: Record<string, { nickname: string; count: number; lastReport: string }> = {};
   164→   164→        for (const r of reports) {
   165→   165→          const key = r.nickname.toLowerCase();
   166→   166→          if (!grouped[key]) {
   167→   167→            grouped[key] = { nickname: r.nickname, count: 0, lastReport: r.data.toISOString() };
   168→   168→          }
   169→   169→          grouped[key].count++;
   170→   170→          if (r.data > new Date(grouped[key].lastReport)) {
   171→   171→            grouped[key].lastReport = r.data.toISOString();
   172→   172→          }
   173→   173→        }
   174→   174→        const ranking = Object.values(grouped)
   175→   175→          .sort((a, b) => b.count - a.count)
   176→   176→          .slice(0, 10);
   177→   177→        return json(ranking);
   178→   178→      }
   179→   179→      default:
   180→   180→        return err("Ação GET desconhecida: " + action);
   181→   181→    }
   182→   182→  } catch (e: unknown) {
   183→   183→    const msg = e instanceof Error ? e.message : "Erro interno";
   184→   184→    return err(msg, 500);
   185→   185→  }
   186→   186→}

   188→   188→export async function POST(req: NextRequest) {
   189→   189→  try {
   190→   190→    const body = await req.json();
   191→   191→    const { action, ...data } = body;
   192→   192→
   193→   193→    if (!action) return err("Campo 'action' é obrigatório");
   194→   194→
   195→   195→    switch (action) {
   196→   196→      // === EMPRÉSTIMOS ===
   197→   197→      case "addEmprestimo": {
   198→   198→        const { player, item, quantidade, dataEmprestimo, tipoMembro, status } = data;
   199→   199→        if (!player || !item || !quantidade) return err("Campos obrigatórios: player, item, quantidade");
   200→   200→        const emp = await db.emprestimo.create({
   201→   201→          data: {
   202→   202→            player,
   203→   203→            item,
   204→   204→            quantidade: Number(quantidade),
   205→   205→            dataEmprestimo: dataEmprestimo ? new Date(dataEmprestimo) : new Date(),
   206→   206→            tipoMembro: tipoMembro || "comum",
   207→   207→            status: status || "pendente",
   208→   208→          },
   209→   209→        });
   210→   210→        return json(emp);
   211→   211→      }
   212→   212→      case "updateEmprestimo": {
   213→   213→        const { id, status, dataPagamento, itemPagamento, quantidadePaga } = data;
   214→   214→        if (!id) return err("id obrigatório");
   215→   215→        const updateData: Record<string, unknown> = {};
   216→   216→        if (status) updateData.status = status;
   217→   217→        if (dataPagamento) updateData.dataPagamento = new Date(dataPagamento);
   218→   218→        if (itemPagamento) updateData.itemPagamento = itemPagamento;
   219→   219→        if (quantidadePaga !== undefined) updateData.quantidadePaga = Number(quantidadePaga);
   220→   220→        const emp = await db.emprestimo.update({
   221→   221→          where: { id },
   222→   222→          data: updateData,
   223→   223→        });
   224→   224→        return json(emp);
   225→   225→      }
   226→   226→
   227→   227→      // === INVESTIDORES ===
   228→   228→      case "addInvestidor": {
   229→   229→        const { nome, observacao } = data;
   230→   230→        if (!nome) return err("nome obrigatório");
   231→   231→        const inv = await db.investidor.create({
   232→   232→          data: { nome, observacao: observacao || null, ordem: 0 },
   233→   233→        });
   234→   234→        return json(inv);
   235→   235→      }
   236→   236→      case "removeInvestidor": {
   237→   237→        const { id } = data;
   238→   238→        if (!id) return err("id obrigatório");
   239→   239→        await db.investidor.delete({ where: { id } });
   240→   240→        return json({ success: true });
   241→   241→      }
   242→   242→      case "reorderInvestidores": {
   243→   243→        const { updates } = data;
   244→   244→        if (!Array.isArray(updates)) return err("updates deve ser array");
   245→   245→        for (const u of updates) {
   246→   246→          await db.investidor.update({
   247→   247→            where: { id: u.id },
   248→   248→            data: { ordem: u.ordem },
   249→   249→          });
   250→   250→        }
   251→   251→        return json({ success: true });
   252→   252→      }
   253→   253→
   254→   254→      // === TABELAS DE TROCA ===
   255→   255→      case "addTabelaTroca": {
   256→   256→        const { itemBase, quantidadeBase, itemResultado, quantidadeResultado, categoria } = data;
   257→   257→        if (!itemBase || !quantidadeBase || !itemResultado || !quantidadeResultado) {
   258→   258→          return err("Campos obrigatórios: itemBase, quantidadeBase, itemResultado, quantidadeResultado");
   259→   259→        }
   260→   260→        const tab = await db.tabelaTroca.create({
   261→   261→          data: {
   262→   262→            itemBase,
   263→   263→            quantidadeBase: Number(quantidadeBase),
   264→   264→            itemResultado,
   265→   265→            quantidadeResultado: Number(quantidadeResultado),
   266→   266→            categoria: categoria || null,
   267→   267→          },
   268→   268→        });
   269→   269→        return json(tab);
   270→   270→      }
   271→   271→      case "removeTabelaTroca": {
   272→   272→        const { id } = data;
   273→   273→        if (!id) return err("id obrigatório");
   274→   274→        await db.tabelaTroca.delete({ where: { id } });
   275→   275→        return json({ success: true });
   276→   276→      }
   277→   277→
   278→   278→      // === TROCAS ===
   279→   279→      case "addTroca": {
   280→   280→        const { player, itemEnviado, quantidadeEnviada, itemRecebido, quantidadeRecebida, tipoMembro, taxaAplicada, lucroBanco } = data;
   281→   281→        if (!player || !itemEnviado || !quantidadeEnviada || !itemRecebido || !quantidadeRecebida) {
   282→   282→          return err("Campos obrigatórios: player, itemEnviado, quantidadeEnviada, itemRecebido, quantidadeRecebida");
   283→   283→        }
   284→   284→        const troca = await db.trocaRegistro.create({
   285→   285→          data: {
   286→   286→            player,
   287→   287→            itemEnviado,
   288→   288→            quantidadeEnviada: Number(quantidadeEnviada),
   289→   289→            itemRecebido,
   290→   290→            quantidadeRecebida: Number(quantidadeRecebida),
   291→   291→            tipoMembro: tipoMembro || "comum",
   292→   292→            taxaAplicada: Number(taxaAplicada) || 0,
   293→   293→            lucroBanco: Number(lucroBanco) || 0,
   294→   294→          },
   295→   295→        });
   296→   296→        return json(troca);
   297→   297→      }
   298→   298→
   299→   299→      // === COMPRAS E VENDAS ===
   300→   300→      case "addCompraVenda": {
   301→   301→        const { tipo, player, item, quantidade, itemPagamento, valor, observacao } = data;
   302→   302→        if (!tipo || !player || !item || !quantidade || valor === undefined) {
   303→   303→          return err("Campos obrigatórios: tipo, player, item, quantidade, valor");
   304→   304→        }
   305→   305→        const cv = await db.compraVenda.create({
   306→   306→          data: {
   307→   307→            tipo,
   308→   308→            player,
   309→   309→            item,
   310→   310→            quantidade: Number(quantidade),
   311→   311→            itemPagamento: itemPagamento || null,
   312→   312→            valor: Number(valor),
   313→   313→            observacao: observacao || null,
   314→   314→          },
   315→   315→        });
   316→   316→        return json(cv);
   317→   317→      }
   318→   318→
   319→   319→      // === CAIXA ===
   320→   320→      case "addCaixa": {
   321→   321→        const { tipo, descricao, item, quantidade, valor, origem } = data;
   322→   322→        if (!tipo || !descricao || !item || !quantidade || !origem) {
   323→   323→          return err("Campos obrigatórios: tipo, descricao, item, quantidade, origem");
   324→   324→        }
   325→   325→        const reg = await db.caixaRegistro.create({
   326→   326→          data: {
   327→   327→            tipo,
   328→   328→            descricao,
   329→   329→            item,
   330→   330→            quantidade: Number(quantidade),
   331→   331→            valor: valor !== undefined ? Number(valor) : null,
   332→   332→            origem,
   333→   333→          },
   334→   334→        });
   335→   335→        return json(reg);
   336→   336→      }
   337→   337→      case "resetCaixa": {
   338→   338→        await db.caixaRegistro.deleteMany();
   339→   339→        await db.emprestimo.deleteMany();
   340→   340→        await db.trocaRegistro.deleteMany();
   341→   341→        await db.compraVenda.deleteMany();
   342→   342→        await db.doador.deleteMany();
   343→   343→        return json({ success: true });
   344→   344→      }
   345→   345→
   346→   346→      // === DOADORES ===
   347→   347→      case "addDoador": {
   348→   348→        const { nome, item, quantidade } = data;
   349→   349→        if (!nome || !item || !quantidade) return err("Campos obrigatórios: nome, item, quantidade");
   350→   350→        const doador = await db.doador.create({
   351→   351→          data: { nome, item, quantidade: Number(quantidade), ordem: 0 },
   352→   352→        });
   353→   353→        return json(doador);
   354→   354→      }
   355→   355→      case "reorderDoadores": {
   356→   356→        const { updates } = data;
   357→   357→        if (!Array.isArray(updates)) return err("updates deve ser array");
   358→   358→        for (const u of updates) {
   359→   359→          await db.doador.update({
   360→   360→            where: { id: u.id },
   361→   361→            data: { ordem: u.ordem },
   362→   362→          });
   363→   363→        }
   364→   364→        return json({ success: true });
   365→   365→      }
   366→   366→
   367→   367→      // === LEILÕES ===
   368→   368→      case "addLeilao": {
   369→   369→        const { donoItem, nomeItem, imagemUrl, quantidade, valorInicial, moedaAceita, taxaCasa, dataExpiracao, tipoOrigem } = data;
   370→   370→        if (!donoItem || !nomeItem || valorInicial === undefined || !moedaAceita || !dataExpiracao) {
   371→   371→          return err("Campos obrigatórios: donoItem, nomeItem, valorInicial, moedaAceita, dataExpiracao");
   372→   372→        }
   373→   373→        const leilao = await db.leilao.create({
   374→   374→          data: {
   375→   375→            donoItem,
   376→   376→            nomeItem,
   377→   377→            imagemUrl: imagemUrl || null,
   378→   378→            quantidade: Number(quantidade) || 1,
   379→   379→            valorInicial: Number(valorInicial),
   380→   380→            moedaAceita,
   381→   381→            taxaCasa: Number(taxaCasa) || 15,
   382→   382→            dataExpiracao: new Date(dataExpiracao),
   383→   383→            tipoOrigem: tipoOrigem || "comum",
   384→   384→          },
   385→   385→        });
   386→   386→        return json(leilao);
   387→   387→      }
   388→   388→      case "updateLeilao": {
   389→   389→        const { id, ...updateData } = data;
   390→   390→        if (!id) return err("id obrigatório");
   391→   391→        const cleaned: Record<string, unknown> = {};
   392→   392→        for (const [k, v] of Object.entries(updateData)) {
   393→   393→          if (v !== undefined && v !== null) {
   394→   394→            if (k === "valorVencedor") cleaned[k] = Number(v);
   395→   395→            else cleaned[k] = v;
   396→   396→          }
   397→   397→        }
   398→   398→        const leilao = await db.leilao.update({ where: { id }, data: cleaned });
   399→   399→        return json(leilao);
   400→   400→      }
   401→   401→      case "removeLeilao": {
   402→   402→        const { id } = data;
   403→   403→        if (!id) return err("id obrigatório");
   404→   404→        await db.leilao.delete({ where: { id } });
   405→   405→        return json({ success: true });
   406→   406→      }
   407→   407→
   408→   408→      // === LANCES ===
   409→   409→      case "addLance": {
   410→   410→        const { leilaoId, jogador, valor } = data;
   411→   411→        if (!leilaoId || !jogador || valor === undefined) {
   412→   412→          return err("Campos obrigatórios: leilaoId, jogador, valor");
   413→   413→        }
   414→   414→        const leilao = await db.leilao.findUnique({ where: { id: leilaoId } });
   415→   415→        if (!leilao) return err("Leilão não encontrado", 404);
   416→   416→        if (leilao.status === "finalizado") return err("Leilão já finalizado");
   417→   417→        // Bloqueia lances em espera quando o tempo +1min já expirou
   418→   418→        if (leilao.status === "espera" && new Date() >= new Date(leilao.dataExpiracao)) {
   419→   419→          return err("O tempo de disputa acabou. Aguardando finalização.");
   420→   420→        }
   421→   421→
   422→   422→        const lances = await db.lance.findMany({
   423→   423→          where: { leilaoId },
   424→   424→          orderBy: { valor: "desc" },
   425→   425→        });
   426→   426→        const maiorLance = lances.length > 0 ? lances[0] : null;
   427→   427→        const valorMinimo = maiorLance ? maiorLance.valor : leilao.valorInicial;
   428→   428→
   429→   429→        if (Number(valor) <= valorMinimo) {
   430→   430→          return err(`O lance deve ser maior que ${valorMinimo} ${leilao.moedaAceita}`);
   431→   431→        }
   432→   432→
   433→   433→        const lance = await db.lance.create({
   434→   434→          data: { leilaoId, jogador, valor: Number(valor) },
   435→   435→        });
   436→   436→
   437→   437→        // Se o tempo original já passou, qualquer lance adiciona +1min e vai para espera
   438→   438→        const now = new Date();
   439→   439→        const originalExpired = now >= new Date(leilao.dataExpiracao);
   440→   440→        if (originalExpired || leilao.status === "espera") {
   441→   441→          const newDeadline = new Date(now.getTime() + 60000);
   442→   442→          await db.leilao.update({
   443→   443→            where: { id: leilaoId },
   444→   444→            data: { dataUltimoLance: new Date(), dataExpiracao: newDeadline, status: "espera" },
   445→   445→          });
   446→   446→        } else {
   447→   447→          await db.leilao.update({
   448→   448→            where: { id: leilaoId },
   449→   449→            data: { dataUltimoLance: new Date() },
   450→   450→          });
   451→   451→        }
   452→   452→
   453→   453→        return json(lance);
   454→   454→      }
   455→   455→
   456→   456→      // === SORTEIOS ===
   457→   457→      case "addSorteio": {
   458→   458→        const { nomeItem, quantidade, duracaoMinutos } = data;
   459→   459→        if (!nomeItem || !quantidade || !duracaoMinutos) {
   460→   460→          return err("Campos obrigatórios: nomeItem, quantidade, duracaoMinutos");
   461→   461→        }
   462→   462→        const dataFim = new Date(Date.now() + Number(duracaoMinutos) * 60 * 1000);
   463→   463→        const sorteio = await db.sorteio.create({
   464→   464→          data: { nomeItem, quantidade: Number(quantidade), duracaoMinutos: Number(duracaoMinutos), dataFim },
   465→   465→        });
   466→   466→        return json(sorteio);
   467→   467→      }
   468→   468→      case "participarSorteio": {
   469→   469→        const { sorteioId, jogador } = data;
   470→   470→        if (!sorteioId || !jogador) return err("Campos obrigatórios: sorteioId, jogador");
   471→   471→        const sorteio = await db.sorteio.findUnique({ where: { id: sorteioId } });
   472→   472→        if (!sorteio) return err("Sorteio não encontrado", 404);
   473→   473→        if (sorteio.status !== "ativo") return err("Sorteio não está ativo");
   474→   474→        if (sorteio.dataFim && new Date(sorteio.dataFim) < new Date()) {
   475→   475→          return err("O tempo do sorteio acabou");
   476→   476→        }
   477→   477→        const existing = await db.participanteSorteio.findFirst({
   478→   478→          where: { sorteioId, jogador },
   479→   479→        });
   480→   480→        if (existing) return err("Você já está participando deste sorteio");
   481→   481→        const p = await db.participanteSorteio.create({ data: { sorteioId, jogador } });
   482→   482→        return json(p);
   483→   483→      }
   484→   484→      case "sortear": {
   485→   485→        const { sorteioId } = data;
   486→   486→        if (!sorteioId) return err("sorteioId obrigatório");
   487→   487→        const participantes = await db.participanteSorteio.findMany({ where: { sorteioId } });
   488→   488→        if (participantes.length === 0) return err("Nenhum participante no sorteio");
   489→   489→        const randomIndex = Math.floor(Math.random() * participantes.length);
   490→   490→        const ganhador = participantes[randomIndex];
   491→   491→        const updated = await db.sorteio.update({
   492→   492→          where: { id: sorteioId },
   493→   493→          data: { status: "finalizado", ganhador: ganhador.jogador, dataFim: new Date() },
   494→   494→        });
   495→   495→        return json({ sucesso: true, ganhador: ganhador.jogador });
   496→   496→      }
   497→   497→      case "removeSorteio": {
   498→   498→        const { id } = data;
   499→   499→        if (!id) return err("id obrigatório");
   500→   500→        await db.sorteio.delete({ where: { id } });
   501→   501→        return json({ success: true });
   502→   502→      }
   503→   503→
   504→   504→      // === LOTÉRICA ===
   505→   505→      case "criarLoterica": {
   506→   506→        const { valorNumero, moedaAceita, premioMinimo, duracaoMinutos } = data;
   507→   507→        if (!valorNumero || !moedaAceita) return err("Campos obrigatórios: valorNumero, moedaAceita");
   508→   508→        // Verificar se ja existe loterica ativa (nao finalizada)
   509→   509→        const ativa = await db.loterica.findFirst({
   510→   510→          where: { status: { in: ["vendas_abertas", "sorteio_realizado"] } },
   511→   511→        });
   512→   512→        if (ativa) return err("Já existe lotérica ativa. Finalize a atual antes de criar uma nova.");
   513→   513→
   514→   514→        const dur = Number(duracaoMinutos) || 60;
   515→   515→        const prem = Number(premioMinimo) || 0;
   516→   516→        const dataFimVendas = new Date(Date.now() + dur * 60 * 1000);
   517→   517→
   518→   518→        // Buscar premio acumulado da ultima loterica sem ganhador
   519→   519→        let acumulado = 0;
   520→   520→        const ultimaSemGanhador = await db.loterica.findFirst({
   521→   521→          where: { ganhador: null, valorPremio: { gt: 0 } },
   522→   522→          orderBy: { dataCriacao: "desc" },
   523→   523→        });
   524→   524→        if (ultimaSemGanhador) {
   525→   525→          acumulado = ultimaSemGanhador.valorPremio || 0;
   526→   526→        }
   527→   527→
   528→   528→        const lot = await db.loterica.create({
   529→   529→          data: {
   530→   530→            valorNumero: Number(valorNumero),
   531→   531→            moedaAceita,
   532→   532→            premioMinimo: prem,
   533→   533→            premioAcumulado: acumulado,
   534→   534→            duracaoMinutos: dur,
   535→   535→            dataFimVendas,
   536→   536→            status: "vendas_abertas",
   537→   537→          },
   538→   538→        });
   539→   539→        // Create 1000 numbers
   540→   540→        const nums = Array.from({ length: 1000 }, (_, i) => ({
   541→   541→          lotericaId: lot.id,
   542→   542→          numero: i + 1,
   543→   543→        }));
   544→   544→        await db.numeroLoterica.createMany({ data: nums });
   545→   545→        return json({ success: true, lotericaId: lot.id, acumulado });
   546→   546→      }
   547→   547→      case "comprarNumero": {
   548→   548→        const { lotericaId, numero, comprador } = data;
   549→   549→        if (!lotericaId || numero === undefined || !comprador) {
   550→   550→          return err("Campos obrigatórios: lotericaId, numero, comprador");
   551→   551→        }
   552→   552→        const lotData = await db.loterica.findUnique({ where: { id: lotericaId } });
   553→   553→        if (!lotData) return err("Lotérica não encontrada", 404);
   554→   554→        if (lotData.status !== "vendas_abertas") return err("Vendas não estão abertas");
   555→   555→        if (lotData.dataFimVendas && new Date(lotData.dataFimVendas) < new Date()) {
   556→   556→          return err("Tempo de vendas encerrado");
   557→   557→        }
   558→   558→        const numEntry = await db.numeroLoterica.findFirst({
   559→   559→          where: { lotericaId, numero: Number(numero) },
   560→   560→        });
   561→   561→        if (!numEntry) return err("Número inválido");
   562→   562→        if (numEntry.comprador) return err("Número já vendido");
   563→   563→
   564→   564→        await db.numeroLoterica.update({
   565→   565→          where: { id: numEntry.id },
   566→   566→          data: { comprador, dataCompra: new Date() },
   567→   567→        });
   568→   568→
   569→   569→        // Arrecadado total aumenta, mas NAO credita no estoque ainda
   570→   570→        const novoArrecadado = (lotData.arrecadadoTotal || 0) + lotData.valorNumero;
   571→   571→        await db.loterica.update({
   572→   572→          where: { id: lotericaId },
   573→   573→          data: { arrecadadoTotal: novoArrecadado },
   574→   574→        });
   575→   575→
   576→   576→        return json({ success: true });
   577→   577→      }
   578→   578→      case "iniciarSorteioLoterica": {
   579→   579→        const { lotericaId } = data;
   580→   580→        if (!lotericaId) return err("lotericaId obrigatório");
   581→   581→        const numeros = await db.numeroLoterica.findMany({ where: { lotericaId } });
   582→   582→        const numerosVendidos = numeros.filter((n) => n.comprador);
   583→   583→        const lotData = await db.loterica.findUnique({ where: { id: lotericaId } });
   584→   584→        if (!lotData) return err("Lotérica não encontrada", 404);
   585→   585→        if (numerosVendidos.length === 0) return err("Nenhum número vendido");
   586→   586→
   587→   587→        const numeroSorteado = Math.floor(Math.random() * 1000) + 1;
   588→   588→        const ganhador = numeros.find((n) => n.numero === numeroSorteado && n.comprador);
   589→   589→
   590→   590→        // Calcular premio: acumulado SUBSTITUI o minimo (nao soma)
   591→   591→        // Ex: minimo 100, acumulado 160 → minimo efetivo = 160
   592→   592→        const effectiveMin = Math.max(lotData.premioMinimo, lotData.premioAcumulado || 0);
   593→   593→        const premio80 = lotData.arrecadadoTotal * 0.8;
   594→   594→        const premioFinal = Math.max(premio80, effectiveMin);
   595→   595→        const taxaBanco = lotData.arrecadadoTotal * 0.2;
   596→   596→
   597→   597→        await db.loterica.update({
   598→   598→          where: { id: lotericaId },
   599→   599→          data: {
   600→   600→            status: "sorteio_realizado",
   601→   601→            numeroSorteado,
   602→   602→            ganhador: ganhador ? ganhador.comprador : null,
   603→   603→            valorPremio: premioFinal,
   604→   604→            dataSorteio: new Date(),
   605→   605→          },
   606→   606→        });
   607→   607→
   608→   608→        // Creditar 20% das vendas no estoque do banco
   609→   609→        if (taxaBanco > 0) {
   610→   610→          await db.caixaRegistro.create({
   611→   611→            data: {
   612→   612→              tipo: "entrada",
   613→   613→              descricao: `Taxa bancária Lotérica (20% das vendas)`,
   614→   614→              item: lotData.moedaAceita,
   615→   615→              quantidade: Math.round(taxaBanco),
   616→   616→              valor: Math.round(taxaBanco),
   617→   617→              origem: "loterica",
   618→   618→            },
   619→   619→          });
   620→   620→        }
   621→   621→
   622→   622→        // Se teve ganhador, registrar saida do premio
   623→   623→        if (ganhador) {
   624→   624→          await db.caixaRegistro.create({
   625→   625→            data: {
   626→   626→              tipo: "saida",
   627→   627→              descricao: `Prêmio Lotérica - Número ${numeroSorteado} (${ganhador.comprador})`,
   628→   628→              item: lotData.moedaAceita,
   629→   629→              quantidade: Math.round(premioFinal),
   630→   630→              valor: Math.round(premioFinal),
   631→   631→              origem: "loterica_premio",
   632→   632→            },
   633→   633→          });
   634→   634→        } else {
   635→   635→          // Ninguem acertou - o premio vira o novo minimo para a proxima
   636→   636→          await db.loterica.update({
   637→   637→            where: { id: lotericaId },
   638→   638→            data: { premioAcumulado: Math.round(premioFinal) },
   639→   639→          });
   640→   640→        }
   641→   641→
   642→   642→        return json({
   643→   643→          success: true,
   644→   644→          numeroSorteado,
   645→   645→          ganhador: ganhador ? ganhador.comprador : null,
   646→   646→          premioFinal: Math.round(premioFinal),
   647→   647→          taxaBanco: Math.round(taxaBanco),
   648→   648→          acumulou: !ganhador,
   649→   649→        });
   650→   650→      }
   651→   651→      case "finalizarLoterica": {
   652→   652→        const { lotericaId } = data;
   653→   653→        if (!lotericaId) return err("lotericaId obrigatório");
   654→   654→        const lotData = await db.loterica.findUnique({ where: { id: lotericaId } });
   655→   655→        if (!lotData) return err("Lotérica não encontrada", 404);
   656→   656→        if (lotData.status !== "sorteio_realizado") return err("Sorteio ainda não foi realizado");
   657→   657→
   658→   658→        // Marca como finalizada - pronto para criar proxima
   659→   659→        await db.loterica.update({
   660→   660→          where: { id: lotericaId },
   661→   661→          data: { status: "finalizada" },
   662→   662→        });
   663→   663→
   664→   664→        return json({ success: true, acumuladoProxima: lotData.premioAcumulado || 0 });
   665→   665→      }
   666→   666→
   667→   667→      // === PRICE REPORTS ===
   668→   668→      case "reportPrice": {
   669→   669→        const { itemId, itemName, nickname, steelQty, steelPrice, cementQty, cementPrice } = data;
   670→   670→        if (!itemId || !itemName || !nickname || steelQty === undefined || cementQty === undefined) {
   671→   671→          return err("Campos obrigatorios: itemId, itemName, nickname, steelQty, cementQty");
   672→   672→        }
   673→   673→        const report = await db.priceReport.create({
   674→   674→          data: {
   675→   675→            itemId,
   676→   676→            itemName,
   677→   677→            nickname,
   678→   678→            steelQty: Number(steelQty),
   679→   679→            steelPrice: Number(steelPrice) || 1,
   680→   680→            cementQty: Number(cementQty),
   681→   681→            cementPrice: Number(cementPrice) || 1,
   682→   682→          },
   683→   683→        });
   684→   684→        return json(report);
   685→   685→      }
   686→   686→
   687→   687→      default:
   688→   688→        return err("Ação POST desconhecida: " + action);
   689→   689→    }
   690→   690→  } catch (e: unknown) {
   691→   691→    const msg = e instanceof Error ? e.message : "Erro interno";
   692→   692→    return err(msg, 500);
   693→   693→  }
